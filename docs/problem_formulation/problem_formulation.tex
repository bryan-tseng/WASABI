\section*{Problem Formulation}
\subsection*{There are a few ways we can formulate the question}
\begin{enumerate}[itemsep=0em, topsep=0.2em]
    \item Similar to Draelos 2020 first identify the binary connections using group testing, then use the identified connections to fit a weighted model.
    \item Directly use a linear-Poisson model to identify weights using group testing. (Works on the assumption that connections are linear)
    \item If not linear, use a neural network or some other nonlinear approach.
\end{enumerate}
We will focus on (2) first, with fixed continuous stimulation parameters (e.g., current, frequency...)

\section*{Goal}
To develop a method for adaptive stimulation that iteratively identifies the most informative stimulation sites. The dynamics of the reach is modeled using a linear-Poisson model , using a weighted linear-Poisson model to identify the most informative stimulation sites and improve the accuracy of the model.

\subsection*{Dynamics model}
We model stimulation as a perturbation to a latent dynamical system through the mapping $\bm{B}$. Specifically, we model the system using a Poisson linear dynamical system (PLDS), which combines firing rate estimation with dimensionality reduction. We assume linear, Markovian (AR(1)) state evolution, Poisson emissions, and stationary $\bm{A}$. Eventually, we can have a Kalman-like update to allow for nonstationarity in $\bm{C}$\phantomsection\label{sec:nonstationarity} and a compositional model for $\bm{A}$
\phantomsection\label{sec:compositional_model}.

\subsubsection*{State space equation}
\vspace{-0.8cm}
\begin{align*}
    \bm{x}_0 &\sim \mathcal{N}(\boldsymbol{\mu}_0, \boldsymbol{\Sigma}_0), \qquad\qquad \bm{u} = \text{softplus}\!\left(\bm{V}\bm{s}_{t_{stim}} + \bm{b}\right) \\
    \bm{x}_{t+1} &= \bm{A}\bm{x}_t + \bm{B}\bm{u}_t + \boldsymbol{\epsilon}_t, \quad \boldsymbol{\epsilon}_t \sim \mathcal{N}(\bm{0}, \bm{Q}) \\
    \bm{y}_t &\sim \text{Poisson}\!\left(\text{softplus}\!\left(\bm{C}\bm{x}_t + \bm{d}\right)\right)
\end{align*}

\section*{Overview}
\begin{enumerate}
    \item \textbf{Control trials}: Primate performs control reaches to the target
    \begin{enumerate}
        \item Fit the Poisson linear dynamical system (PLDS) using Extended Kalman Filter EKF to the control reaches to obtain $\hat{\bm{\theta}}_{\mathrm{ctrl}}$
    \end{enumerate}
    \item \textbf{Stim trials}:
    \begin{enumerate}
        \item \textbf{Model fitting}: Between each trial, we fit the spike counts $\{\bm{y}_{i,1:T}\}$ to the model using the EKF + RTS smoothing to get $\{\hat{\bm{x}}_{i,1:T}, \bm{P}_{i,1:T}\}$. We also get updated parameters $\hat{\bm{\theta}}_{\mathrm{i}} \leftarrow \hat{\bm{\theta}}_{\mathrm{i-1}}$
        \item At the first trial, we have no information about stim-to-latent mapping, so the first stimulation sites are chosen at random.
    \end{enumerate}
    \item \textbf{Acquisition:}
    \begin{enumerate}
        \item \textbf{Recursive least squares}: Using RLS to track $\hat{\bm{B}}_i$ and $\boldsymbol{\hat\Sigma}_{B_{i|i}}$ gives us an estimate of the covariance in the stim-to-latent mapping.
        \item \textbf{Stim selection}: Choose the next stimulation $\bm{s}_{i+1}$ using the covariance estimate of $\hat{\bm{B}}_{i+1|i}$ ($\hat{\bm{\Sigma}}_{B_{i+1|i}}$)
    \end{enumerate}
    \item Perform the next stimulation and repeat 2-3.
\end{enumerate}

\section*{Long overview}
\begin{enumerate}
    \item \textbf{Control trials}: Monkey performs control reaches to the target
    \item We fit the PLDS to the control reaches to obtain $\hat{\bm{A}}_{\mathrm{ctrl}}, \hat{\bm{C}}_{\mathrm{ctrl}}, \hat{\bm{d}}_{\mathrm{ctrl}}, \hat{\bm{Q}}_{\mathrm{ctrl}}$
    \item \textbf{Stim trials}: During each trial, perform online filtering using the fitted parameters $\hat{\bm{\theta}}_{\mathrm{i-1}}$,  to get $\{\bm{x}_{i,1:t}\}$ in real time. This gives us the state at which the stimulation happens.
    \begin{enumerate}
        \item At the first trial, we initialize with $\hat{\bm{\theta}}_{\mathrm{i-1}} = \{\hat{\bm{A}}_{\mathrm{ctrl}}, \hat{\bm{B}}_{0|0}, \hat{\bm{b}}_{0|0}, \hat{\bm{C}}_{\mathrm{ctrl}}, \hat{\bm{d}}_{\mathrm{ctrl}}, \hat{\bm{Q}}_{\mathrm{ctrl}}, \hat{\bm{\mu}}_{0}, \hat{\bm{\Sigma}}_{0}\}$, where $\hat{\bm{B}}_{0|0} = \bm{0}$, $\hat{\bm{\Sigma}}_{B_{0|0}} = \sigma_B^2\,\bm{I}$, and $\hat{\bm{b}}_{0|0} = \bm{0}$.
        \item This means that we have no information about the stim-to-latent mapping or the perturbation threshold and the first stimulation sites are chosen at random.
    \end{enumerate}
    \item \hyperref[sec:model_fitting]{\textbf{Model fitting}}: Between each trial, we fit the spike counts $\{\bm{y}_{i,1:T}\}$ to the model using the Extended Kalman Filter (EKF)-style integral and stochastic gradient descent (SGD). This gives us updated parameters $\hat{\bm{\theta}}_{\mathrm{i}} \leftarrow \hat{\bm{\theta}}_{\mathrm{i-1}}$
    \begin{enumerate}
        \item The previous trial would "warm-start" the convergence of the next trial, so we only need to run a few iterations of EKF and SGD between each trial to get a converging estimate of the parameters.
    \end{enumerate}
    \item \textbf{Acquisition:}
    \begin{enumerate}
        \item \hyperref[sec:RLS]{\textbf{Recursive least squares}}: Using RLS to track $\hat{\bm{B}}_i$ and $\boldsymbol{\hat\Sigma}_{B_{i|i}}$ gives us an estimate of the covariance in the stim-to-latent mapping. This step uses $\hat{\bm{A}}_i, \hat{\bm{Q}}_i, \hat{\bm{x}}_{i,t|T}, \bm{P}_{i,t|T}$, and $\hat{\bm{d}}_i$ estimated from  steps 4.
        \item \hyperref[sec:stim_selection]{\textbf{Stim selection}}: Choose the next stimulation $\bm{s}_{i+1}$ using the covariance estimate of $\hat{\bm{B}}_{i+1|i}$ ($\hat{\bm{\Sigma}}_{B_{i+1|i}}$), which is the uncertainty in the stim-to-latent mapping after trial $i+1$ after a given stimulation.
    \end{enumerate}
    \item Perform the next stimulation and repeat 3-5.
\end{enumerate}

\begin{minipage}[t]{0.47\textwidth}
\textbf{First stim trial initialization} ($i = 1$)\\[0.5em]
Warm start with fitted control parameters:

    \hat{\bm{d}}_1 &= \log\bar{\bm{y}}_{1}

\begin{align*}
    \hat{\bm{B}}_{0|0} &= \bm{0}_{K \times N} \\
    \hat{\bm{\Sigma}}_{B_{0|0}}^{(k)} &= \sigma_k^2\,\bm{I}_N \\

    \item $\sigma_C^2$: controls prior uncertainty over the latent-to-observation mapping

    \item a spherical prior on each row means $\bm{s}_1$ is chosen at random
\end{itemize}

\subsection*{Online forward filtering during a trial}
\phantomsection{\label{sec:online_filtering}}
During a trial, we estimate the latent state $\bm{x}_{i,t}$ in real time using the EKF forward pass with the current parameter estimates $\hat{\boldsymbol{\theta}}_{i-1}$. This gives us an estimate of the latent state in real time.
\subsection*{EKF forward pass} %  (Ignore Ds for now)
The marginal likelihood of the latent trajectory is:
\begin{align*}
    \log p(\bm{y}_{1:T}|\boldsymbol{\theta})
        &= \log \int p(\bm{y}_{1:T} \mid \bm{x}_{1:T};\boldsymbol{\theta})\,
                       p(\bm{x}_{1:T};\boldsymbol{\theta})\, d\bm{x}_{1:T}\\
\bm{y}_t &\sim \text{Poisson}\!\left(\text{softplus}\!\left(\bm{C}\bm{x}_t + \bm{d}\right)\right)\\
\boldsymbol{\theta} &= \{\bm{A, B, b, C, d, Q, \mu_0, \Sigma_0}\}
\end{align*}

Because the Poisson emission $p(\bm{y}_{1:t} \mid \bm{x}_{1:t};\boldsymbol{\theta})$
is non-Gaussian, the integral is intractable. At time $t$, we need to approximate it by linearizing the emission around $\hat{\bm{x}}_{t+1|t}$, the current filtered mean (latent state), yielding a Gaussian approximation to the predictive distribution
\begin{align*}
    p(\bm{y}_t \mid \bm{y}_{1:t};\boldsymbol{\theta})
        \approx \mathcal{N}(\boldsymbol{\lambda}_{t+1|t},\, \bm{S}_t), \qquad
    \boldsymbol{\lambda}_{t+1|t} = \text{softplus}\!\left(\bm{C}\hat{\bm{x}}_{t+1|t} + \bm{d}\right)
\end{align*}
where $\bm{S}_t$ is the estimated covariance.
\begin{align*}
    \mathcal{L}(\boldsymbol{\theta})
        \approx \sum_{t=1}^{T} \log \mathcal{N}(\bm{y}_t|\,
                \boldsymbol{\lambda}_{t+1|t},\, \bm{S}_t)
\end{align*}

\subsection*{Emission Jacobian}
In the EKF formulation, we approximate the observation operator $\bm{H}_t$ with the Emission Jacobian described here:
For $\boldsymbol{\lambda}_t = \text{softplus}(\bm{C}\hat{\bm{x}}_t + \bm{d})$, we have the Jacobian (log-rate with respect to the latent state) is:
\begin{align*}
    \bm{H}_t &= \frac{\partial \boldsymbol{\lambda}_t}{\partial \bm{x}_t}\Bigg|_{\hat{\bm{x}}_t}
                 = \text{diag}(\sigma(\bm{C}\hat{\bm{x}}_t + \bm{d}))\,\bm{C}
                 \in \mathbb{R}^{N \times K}
\end{align*}
where $\sigma(\cdot)$ is the sigmoid function (derivative of softplus). We calculate the Emission Jacobian $\bm{H}_t$ to linearize the non-Gaussian Poisson emission and plug into the Kalman equations. This propagates the Gaussian estimations of latent states $\bm{x}_t$ one time step forward each, which we can plug in the Kalman equations.
\subsection*{EKF forward pass prediction update}
\noindent
\begin{minipage}[t]{0.47\textwidth}
\begin{align*}
    \hat{\bm{x}}_{t|t-1} &= \bm{A}\hat{\bm{x}}_{t-1|t-1} 
    + \bm{B}\,\text{softplus}\!\left(\bm{V}\bm{s}_{t-1} + \bm{b}\right) \\
    \bm{P}_{t|t-1}       &= \bm{A}\bm{P}_{t-1|t-1}\bm{A}^\top + \bm{Q} \\
    \bm{S}_t             &= \bm{H}_t \bm{P}_{t|t-1} \bm{H}_t^\top
                                + \text{diag}(\boldsymbol{\lambda}_{t|t-1}) \\
    \bm{K}_t             &= \bm{P}_{t|t-1}\bm{H}_t^\top \bm{S}_t^{-1} \\
    \hat{\bm{x}}_{t|t}   &= \hat{\bm{x}}_{t|t-1}
                                + \bm{K}_t(\bm{y}_t - \boldsymbol{\lambda}_{t|t-1}) \\
    \bm{P}_{t|t}         &= (\bm{I} - \bm{K}_t \bm{H}_t)\bm{P}_{t|t-1}
\end{align*}
\end{minipage}
\hfill\vline\hfill
\begin{minipage}[t]{0.47\textwidth}
\begin{itemize}[leftmargin=*, itemsep=0em, topsep=0.2em]
    \item $\hat{\bm{x}}_{t|t-1} \in \mathbb{R}^{K}$: predicted latent state mean
    \item $\hat{\bm{x}}_{t|t} \in \mathbb{R}^{K}$: filtered latent state mean
    \item $\bm{P}_{t|t-1} \in \mathbb{R}^{K \times K}$: predicted latent state covariance
    \item $\bm{P}_{t|t} \in \mathbb{R}^{K \times K}$: filtered latent state covariance
    \item $\bm{S}_t \in \mathbb{R}^{N \times N}$: innovation covariance
    \item $\bm{K}_t \in \mathbb{R}^{K \times N}$: Kalman gain
    \item $(\bm{y}_t - \boldsymbol{\lambda}_{t|t-1})$: innovation (residual)
    \item $\bm{Q} \in \mathbb{R}^{K \times K}$: latent process noise covariance
    \item $\text{diag}(\boldsymbol{\lambda}_{t|t-1})$: Poisson observation noise
    \item $\bm{H}_t \in \mathbb{R}^{N \times K}$: locally linear approximation of emission matrix
\end{itemize}
\end{minipage}

\subsection*{Offline parameter update (Outline)}
At the end of each trial $i$, the following steps are performed in order:
\begin{enumerate}
    \item ``E step'' \textbf{EKF forward pass + RTS smoother}:
    Run the EKF forward over $\bm{y}_{i,1:T}$ with $\hat{\bm{\theta}}_{i-1}$ fixed, then run the RTS smoother backward to get filtered means and covariances $\{\hat{\bm{x}}_{i,1:T},\, \bm{P}_{i,1:T}\}$. $\{\hat{\bm{x}}_{i,t|t},\, \bm{P}_{i,t|t}\}_{t=1}^{T}$ is the lag-one cross-covariance needed for the M-steps below.
    \item ``M step'' \textbf{Factorized estimation of parameters}: \phantomsection\label{sec:factorized_estimation} We factorize the estimation of parameters to get point estimates of parameters with no closed form ($\hat{\bm{C}}_i$, $\hat{\bm{d}}_i$, $\hat{\bm{b}}_i$), but still getting covariance estimates for those that we can using EM. This also keeps the \href{https://probml.github.io/dynamax/notebooks/hmm/casino_hmm_learning.html#compare-the-learning-algorithms}{faster convergence rate of the EM algorithm}.
    \begin{enumerate}
        \item \hyperref[sec:sgd_parameter_update]{\textbf{SGD update $\hat{\bm{\theta}}_i$}}
            Take one gradient step on the single-trial EKF marginal log-likelihood
            $\mathcal{L}_i = \sum_{t=1}^{T} \log\mathcal{N}(\bm{y}_{i,t} \mid
            \boldsymbol{\lambda}_{i,t|t-1}, \bm{S}_{i,t})$:
            \begin{align*}
                \hat{\bm{\theta}}_i \leftarrow \hat{\bm{\theta}}_{i-1} - \eta_i\, \nabla_{\bm{C},\bm{d}}\,
                    \bigl[-\mathcal{L}_i(\hat{\boldsymbol{\theta}}_{i-1})\bigr]
            \end{align*}
    \end{enumerate}
    \item \textbf{Repeat 1 and 2 until convergence}:
    \begin{enumerate}
        \item Steps (a)-(c) are handled in the DYNAMAX package using conditional moments gaussian smoother (CMGS), but we lay out the details below.
        \item We may need \href{https://probml.github.io/dynamax/notebooks/hmm/casino_hmm_learning.html#hyperparameter-tuning}{hyperparameter optimization} on the batch size and learning rate.
        \item This process can terminate early if the next trial happens before convergence.
        \item We will need to analyze our \textbf{convergence criterion}\phantomsection\label{sec:convergence_criterion} and if we need an algorithm that is quicker / more accurate. (Low rank prior of loading matrix, instead of a state-space formulation) %we may consider online variational inference or ..
    \end{enumerate}
\end{enumerate}

\subsection*{EKF forward pass + RTS smoother}
The EKF forward pass is identical to the online version. 
(see \hyperref[sec:online_filtering]{\textbf{Online forward filtering}})

\subsubsection*{RTS smoother backward pass}
\begin{align*}
    \bm{G}_t             &= \bm{P}_{t|t}\bm{A}^\top \bm{P}_{t+1|t}^{-1} \\
    \hat{\bm{x}}_{t|T}   &= \hat{\bm{x}}_{t|t}
                                + \bm{G}_t\!\left(\hat{\bm{x}}_{t+1|T} - \hat{\bm{x}}_{t+1|t}\right) \\
    \bm{P}_{t|T}         &= \bm{P}_{t|t}
                                + \bm{G}_t\!\left(\bm{P}_{t+1|T} - \bm{P}_{t+1|t}\right)\bm{G}_t^\top
\end{align*}
initialized at $\hat{\bm{x}}_{T|T}$, $\bm{P}_{T|T}$ and run backward for $t = T-1, \ldots, 1$.

\subsection*{SGD parameter update $\hat{\bm{\theta}}_i$}
\phantomsection\label{sec:sgd_parameter_update}
Parameters related to the Poisson observation equation ($\hat{\bm{C}}_i$, $\hat{\bm{d}}_i$) and the stim-to-latent threshold ($\hat{\bm{b}}_i$) have no closed form. So, we made a design choice to update the parameter set by using SGD and autodiff through the posterior:
\begin{align*}
    \boldsymbol{\theta}^{(\tau+1)}
        &= \boldsymbol{\theta}^{(\tau)}
           - \eta_\tau \,\nabla_{\boldsymbol{\theta}}\,
             \bigl[-\mathcal{L}(\boldsymbol{\theta}^{(\tau)})\bigr]
\end{align*}
where $\eta_\tau$ is the effective learning rate defined by the Adam optimizer. More specifically:
\begin{align*}
    \bm{m}_{\tau} &= \beta_1 \bm{m}_{\tau-1}
                         + (1-\beta_1)\,\nabla_{\boldsymbol{\theta}}
                         \bigl[-\mathcal{L}(\boldsymbol{\theta}^{(\tau)})\bigr]
                         \tag{momentum} \\
    \bm{v}_{\tau} &= \beta_2 \bm{v}_{\tau-1}
                         + (1-\beta_2)\,\nabla_{\boldsymbol{\theta}}
                         \bigl[-\mathcal{L}(\boldsymbol{\theta}^{(\tau)})\bigr]^2
                         \tag{RMSprop} \\
    \hat{\bm{m}}_{\tau} &= \frac{\bm{m}_{\tau}}{1-\beta_1^\tau}, \qquad
    \hat{\bm{v}}_{\tau}  = \frac{\bm{v}_{\tau}}{1-\beta_2^\tau}
                         \tag{bias correction} \\
    \boldsymbol{\theta}^{(\tau+1)}
                      &= \boldsymbol{\theta}^{(\tau)}
                         - \frac{\alpha}{\sqrt{\hat{\bm{v}}_{\tau}}+\epsilon}\,
                         \hat{\bm{m}}_{\tau}
                         \tag{parameter update}
\end{align*}
where $\bm{m}_\tau$ and $\bm{v}_\tau$ are the first and second moment
estimates, $\beta_1 = 0.9$, $\beta_2 = 0.999$, $\epsilon = 10^{-8}$ are the
default hyperparameters, and $\alpha$ is the base learning rate. Gradients are computed via automatic differentiation in JAX in the DYNAMAX package. We get $\hat{\bm{\theta}}_i$ after convergence.

\clearpage
\subsection*{Getting the posterior $p(\hat{\bm{B}}_{i|i-1} \mid \bm{y}_{1:i-1}, \bm{s}_{i})$}
Stimulation pattern $\bm{s}_i \in \{0,1\}^N$ is fixed for trial $i$ and happens at $t_{\mathrm{stim}}$.
\begin{align*}
    \bm{x}_{i,0} &\sim \mathcal{N}(\boldsymbol{\mu}_0, \boldsymbol{\Sigma}_0) \\
    \bm{x}_{i, t} &= \bm{A}\bm{x}_{i, t-1} + \bm{B}\bm{u}_i + \boldsymbol{\epsilon}_t, \quad \boldsymbol{\epsilon}_t \sim \mathcal{N}(\bm{0}, \bm{Q}) \\
    \bm{y}_{i, t} &\sim \text{Poisson}\!\left(\text{softplus}\!\left(\bm{C}\bm{x}_{i, t} + \bm{d}\right)\right)
\end{align*}
We assume we can factorize across the $\boldsymbol{\theta}$s (Much like coordinate ascent variational inference). We get the estimates of $\boldsymbol{\theta} \setminus \bm{B}$ known from PLDS fit via EKF SGD from trials $1, \ldots, i-1$ and treated as fixed at trial $i$. $\bm{s}_{t}$ is known:
\begin{align*}
    \boldsymbol{\theta} \setminus \bm{B} = \{\bm{A},\,\bm{C},\,\bm{d},\,
    \bm{Q},\,\boldsymbol{\mu}_0,\,\boldsymbol{\Sigma}_0\}
    = \hat{\boldsymbol{\theta}}_{i-1} \setminus \bm{B}
\end{align*}

\begin{align*}
    p(\boldsymbol{\theta} \mid \bm{y}_{1:t})
        &\propto p(\bm{y}_{1:t} \mid \boldsymbol{\theta}) \cdot p(\boldsymbol{\theta}) \\
    p(\bm{B} \mid \bm{y}_{1:t}, \bm{s}_{t})
        &\propto p(\bm{y}_{1:t} \mid \bm{B},\, \boldsymbol{\theta} \setminus \bm{B}) \cdot
                 p(\bm{B}) \cdot p(\bm{A}) \cdot p(\bm{C}) \cdots
                 \quad \text{(by independence of } \boldsymbol{\theta} \text{ components)}
\end{align*}

Absorbing $p(\boldsymbol{\theta} \setminus \bm{B})$ into the constant and expanding over $\bm{x}_{1:t}$:
\begin{equation}
    p(\bm{B} \mid \bm{y}_{1:t}, \bm{s}_{t}) \propto 
        \underbrace{p(\bm{B})}_{\text{Initialize via }\mathcal{N}(\bm{0}, \boldsymbol{\Sigma}_0^B)} 
        \int 
        \underbrace{p(\bm{y}_{1:t} \mid \bm{x}_{1:t}, \boldsymbol{\bm{B}, \theta} \setminus \bm{B})}_{\text{Poisson emission}} \;
        \underbrace{p(\bm{x}_{1:t} \mid \bm{s}_{t}, \boldsymbol{\theta})}_{\text{depends on } \bm{B}, \text{can get from EKF}} 
        \, d\bm{x}_{1:t}
       \label{eq:posterior_B}
\end{equation}
Likelihood after marginalizing out $\bm{x}_{1:t}$:
\begin{align*}
    p(\bm{y}_{1:t} \mid \boldsymbol{\theta}) &= \int 
        \underbrace{p(\bm{y}_{1:t} \mid \bm{x}_{1:t}, \boldsymbol{\theta})}_{\text{Poisson emission, need EKF approx.}} \;
        \underbrace{p(\bm{x}_{1:t} \mid \bm{s}_{t}, \boldsymbol{\theta})}_{\text{Gaussian dynamics prior}} 
        \, d\bm{x}_{1:t}
\end{align*}
\begin{align*}
    \boldsymbol{\theta} &= \{\bm{A},\,\bm{B},\,\bm{C},\,
                             \bm{d},\,\bm{Q},\,\boldsymbol{\mu}_0,\,\boldsymbol{\Sigma}_0\}
\end{align*}

Bayes':
\begin{align*}
    \underbrace{p(\boldsymbol{\theta} \mid \bm{y}_{1:t})}_{\text{posterior}} &= \frac{
        \overbrace{p(\bm{y}_{1:t} \mid \boldsymbol{\theta})}^{\text{likelihood}} \cdot
        \overbrace{p(\boldsymbol{\theta})}^{\text{prior}}
    }{
        \underbrace{p(\bm{y}_{1:t})}_{\text{evidence, constant in } \boldsymbol{\theta}}
    }
\end{align*}

\section*{Acquisition}
\phantomsection\label{sec:RLS}

We select the next stimulation pattern by minimizing either the determinant (D-optimal) or the trace (A-optimal) of the covariance of $\bm{B}$. From SGD, we only get point estimates of the PLDS parameters. Therefore, we run a separate recursive least squares (RLS) tracker for $\bm{B}$. We work with two representations (assumptions) of that posterior covariance:
\begin{enumerate}
    \item \textbf{Per row covariance}: $K$ separate $N\times N$ matrices $\hat{\bm{\Sigma}}_{B_i}^{(k)}$, one per latent, assuming stimulation effects on the latents do not interact. We use this for \textbf{D-optimal}, since the determinant calculation for the full covariance may be intractable. The per row log-determinant decomposes into a sum of $K$ terms.
    \item \textbf{Full $NK\times NK$ covariance} over $\bm{B}$ stacked into a vector. Retains cross-latent correlations. We use this for \textbf{A-optimal}, which doesn't need a determinant of a
    large matrix.
    \begin{enumerate}
        \item TODO: The trace reads only diagonal entries, but its one-trial-ahead \emph{value} still depends on the off-diagonal blocks through the update, so the two pairings are not interchangeable.
    \end{enumerate}
\end{enumerate}


\subsection*{One-step ahead prediction of latents gives us the estimate of the perturbation}

Given the smoothed latents $\{\hat{\bm{x}}_{i,t|T}\}_{t=1}^T$, we can treat each row as a scalar observation RLS. We describe it in the following: 

The effective stimulation input is $\bm{u}_i = \mathrm{softplus}(\bm{V}\bm{s}_i+\bm{b}) \in \mathbb{R}^N$. The stimulation-induced perturbation is $\bm{z}_i = \bm{B}\bm{u}_i$, and from the EKF forward pass + RTS smoother at trial $i$ we estimate it by the one-step residual. Here, $t$ denotes the final time bin at which the stimulation input is applied, so the stimulation-induced perturbation is observed in the transition from $t$ to $t+1$.
\begin{align*}
    \hat{\bm{z}}_i &:= \hat{\bm{x}}_{i,t+1|T} - \hat{\bm{A}}_i\hat{\bm{x}}_{i,t|T}\\
    \bm{z}_i &= \bm{B}\bm{u}_i\\
    \bm{z}_i &= \hat{\bm{x}}_{i,t+1|T} - \hat{\bm{A}}_i\hat{\bm{x}}_{i,t|T} - \boldsymbol{\varepsilon}_i, \quad \boldsymbol{\varepsilon}_i \sim \mathcal{N}(\bm{0},\bm{R}_i),
\end{align*}
The error covariance collects the smoother uncertainty at both timesteps plus the process noise:
\begin{align*}
    \bm{R}_i
    &= \mathrm{Cov}(\hat{\bm{x}}_{i,t+1|T}) + \hat{\bm{A}}_i\mathrm{Cov}(\hat{\bm{x}}_{i,t|T})\hat{\bm{A}}_i^\top + \hat{\bm{Q}}_i \\
    &\quad - \mathrm{Cov}(\hat{\bm{x}}_{i,t+1|T}, \hat{\bm{x}}_{i,t|T})\hat{\bm{A}}_i^\top
           - \hat{\bm{A}}_i\mathrm{Cov}(\hat{\bm{x}}_{i,t+1|T}, \hat{\bm{x}}_{i,t|T})^\top \\ &\text{(Assumption: we don't have cross-cov, drop for now, and we ignore $\mathrm{Cov}(\hat{\bm{A}})$)} \\
    &\approx \hat{\bm{P}}_{i,t+1|T} + \hat{\bm{A}}_i\hat{\bm{P}}_{i,t|T}\hat{\bm{A}}_i^\top + \hat{\bm{Q}}_i
\end{align*}

\inlinecomment{The RTS smoother does produce the lag-one cross-covariance (it is already needed for the M-step). If it is available, keep these two terms rather than dropping them --- they are subtractive, so dropping them inflates $\bm{R}_i$ and makes every stimulus look less informative than it is.}

\subsection*{Multiple step ahead perturbation gives us perturbations propagated by the dynamics operator}

If stimulation happens over multiple time bins where we have missing observations, we treat the prestim and poststim observation as a multi-step propagation. The stimulation-induced perturbation is propagated through the dynamics operator over the stimulation duration. Let $\tau$ denote the number of stimulation bins. Then
\begin{align*}
    \bm{M}_{\mathrm{\tau}}
    &:=
    \bm{I}
    + \bm{A}
    + \cdots
    + \bm{A}^{\mathrm{\tau}-1}.
\end{align*}

The effective perturbation accumulated over the stimulation period is therefore
\begin{align*}
    \bm{z}_i
    &=
    \bm{M}_{\mathrm{\tau}}\bm{B}\bm{u}_i.
\end{align*}

If $\hat{\bm{x}}_{i,t|T}$ denotes the smoothed latent immediately before the stimulation period and $\hat{\bm{x}}_{i,t+\tau|T}$ denotes the smoothed latent immediately after stimulation, the one-step residual is replaced by the multi-step residual
\begin{align*}
    \hat{\bm{z}}_i
    &:=
    \hat{\bm{x}}_{i,t+\tau|T}
    -
    \hat{\bm{A}}_i^{\mathrm{\tau}}
    \hat{\bm{x}}_{i,t|T}
\end{align*}

Thus,
\begin{align*}
    \bm{z}_i
    &=
    \bm{M}_{\mathrm{\tau}}\bm{B}\bm{u}_i,
    \\
    \bm{z}_i
    &=
    \hat{\bm{x}}_{i,t+\tau|T}
    -
    \hat{\bm{A}}_i^{\mathrm{\tau}}
    \hat{\bm{x}}_{i,t|T}
    -
    \boldsymbol{\varepsilon}_i,
    \qquad
    \boldsymbol{\varepsilon}_i
    \sim
    \mathcal{N}(\bm{0},\bm{R}_i).
\end{align*}
The error covariance collects the smoother uncertainty at both endpoints plus the process noise accumulated over the stimulation period:
\begin{align*}
    \bm{R}_i
    &=
    \mathrm{Cov}(\hat{\bm{x}}_{i,t+\tau|T})
    +
    \hat{\bm{A}}_i^{\mathrm{\tau}}
    \mathrm{Cov}(\hat{\bm{x}}_{i,t|T})
    \left(\hat{\bm{A}}_i^{\mathrm{\tau}}\right)^\top
    \\
    &\quad
    +
    \sum_{j=0}^{\mathrm{\tau}-1}
    \hat{\bm{A}}_i^j
    \hat{\bm{Q}}_i
    \left(\hat{\bm{A}}_i^j\right)^\top
    \\
    &\quad
    - \text{cross-covariance terms}
\end{align*}

Under the assumption that we do not have the required cross-covariances, and ignoring $\mathrm{Cov}(\hat{\bm{A}}_i)$,
\begin{align*}
    \bm{R}_i
    &\approx
    \hat{\bm{P}}_{i,t+\tau|T}
    +
    \hat{\bm{A}}_i^{\mathrm{\tau}}
    \hat{\bm{P}}_{i,t|T}
    \left(\hat{\bm{A}}_i^{\mathrm{\tau}}\right)^\top
    \\
    &\quad
    +
    \sum_{j=0}^{\mathrm{\tau}-1}
    \hat{\bm{A}}_i^j
    \hat{\bm{Q}}_i
    \left(\hat{\bm{A}}_i^j\right)^\top
\end{align*}

\subsection*{D-optimal (per row)}

D-optimal experimental design minimizes the determinant of the covariance of $\bm{B}$ (the volume of the uncertainty ellipsoid). Here we take the per-row representation: the rows of the posterior $\bm{B}$ are assumed to be row-independent, with covariance $\bm{\Sigma}_{B_i} = \operatorname{diag}(\bm{\Sigma}_{B_i}^{(1)}, \bm{\Sigma}_{B_i}^{(2)}, \dots, \bm{\Sigma}_{B_i}^{(k)})$, so each latent carries its own $N\times N$ covariance and the log-determinant decomposes into a sum of $K$ terms.

\subsubsection*{Row decomposition and priors}
Let $\bm{w}^{(k)} \in \mathbb{R}^{N}$ be:
\begin{align*}
    \bm{B} = \bm{W}^\top \in \mathbb{R}^{K\times N} =
    \begin{bmatrix}\bm{w}_1^\top \\ \bm{w}_2^\top \\ \vdots \\ \bm{w}_K^\top\end{bmatrix}.
\end{align*}
Row $k$ collects the effect of stimulation at each of the $N$ channels on latent $k$ alone. Each row has its own prior,
\begin{align*}
    \bm{w}^{(k)} \sim \mathcal{N}\big(\bm{0},\, \hat{\bm{\Sigma}}_{B_0}^{(k)}\big),
    \qquad \hat{\bm{\Sigma}}_{B_0}^{(k)} = \sigma_k^2\,\bm{I}_N,
\end{align*}
isotropic across channels, $\sigma_k^2$ is the prior "scale" for how strongly stimulation is expected to move latent $k$. We initialize this to be proportional to the ``magnitude'' of row $k$ of
$\bm{B}$. Setting all $\sigma_{B, k}^2 = \sigma_B^2$ recovers a shared variance scale. Treating the rows as independent is the assumption that stimulation effects on each latent do not interact. The only coupling that remains enters through the shared noise covariance $\bm{R}_i$.

\subsubsection*{Predictive covariance of the observation}
For row $k$, the relevant predictive quantity is the variance of the scalar $\bm{w}^{(k)\top}\bm{u}_i$ under the current prior $\bm{w}^{(k)}\sim\mathcal N(\hat{\bm{w}}_{i-1}^{(k)},\hat{\bm{\Sigma}}_{B_{i-1}}^{(k)})$. The variance of this predictive quantity is:
\begin{align*}
    \mathrm{Var}(\bm{w}^{(k)\top}\bm{u}_i)
        &= \mathbb{E}\!\left[(\bm{w}^{(k)\top}\bm{u}_i)^2\right] - \big(\mathbb{E}[\bm{w}^{(k)\top}\bm{u}_i]\big)^2 \\
        &= \mathbb{E}\!\left[\bm{u}_i^\top \bm{w}^{(k)}\bm{w}^{(k)\top} \bm{u}_i\right] - \bm{u}_i^\top\,\mathbb{E}[\bm{w}^{(k)}]\mathbb{E}[\bm{w}^{(k)}]^\top\,\bm{u}_i \\
        &= \bm{u}_i^\top\Big(\mathbb{E}\!\left[\bm{w}^{(k)}\bm{w}^{(k)\top}\right] - \mathbb{E}[\bm{w}^{(k)}]\mathbb{E}[\bm{w}^{(k)}]^\top\Big)\bm{u}_i
         = \bm{u}_i^\top\,\mathrm{Cov}(\bm{w}^{(k)})\,\bm{u}_i.
\end{align*}
Using $\mathrm{Cov}(\bm{w}^{(k)})=\hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}$ from the prior, this gives
\begin{align*}
    \sigma^2_{k|\bm{u}_i} := \mathrm{Var}(\bm{w}^{(k)\top}\bm{u}_i) = \bm{u}_i^\top \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)} \bm{u}_i,
\end{align*}

Without multi-step propagation, these row-wise uncertainties enter different coordinates of $\bm{z}_i$ independently. With multi-step propagation, however, the dynamics operator mixes these latent coordinates. Specifically,

\begin{align*}
    \bm{z}_i
    =
    \bm{M}_{\mathrm{\tau}}
    \bm{B}\bm{u}_i.
\end{align*}

Therefore the parameter-uncertainty contribution to the predictive covariance is

\begin{align*}
    \bm{D}_i
    &:=
    \bm{M}_{\mathrm{\tau}}
    \operatorname{diag}
    \left(
        \sigma^2_{1|\bm{u}_i},
        \dots,
        \sigma^2_{K|\bm{u}_i}
    \right)
    \bm{M}_{\mathrm{\tau}}^\top.
\end{align*}

Adding the observation noise gives the full predictive covariance

\begin{align*}
    \mathbb{E}[\bm{z}_i\mid\bm{u}_i]
    &=
    \bm{M}_{\mathrm{\tau}}
    \bm{B}\bm{u}_i,
    \\
    \mathrm{Var}(\bm{z}_i\mid\bm{u}_i)
    &=
    \bm{S}_i
    :=
    \bm{D}_i+\bm{R}_i
    \in\mathbb{R}^{K\times K}.
\end{align*}
The two components are:
\begin{enumerate}[itemsep=0em, topsep=0.2em]
    \item $\bm{D}_i$: uncertainty in the prediction of $\bm{z}_i$ due to uncertainty in $\bm{B}$. The covariance is diagonal before propagation, but $\bm{M}_{\mathrm{\tau}}$ can introduce correlations across latent coordinates.
    \item $\bm{R}_i$: uncertainty in the observation of $\bm{z}_i$ from noise in the measurement process, inherited from the latents. This noise can correlate across latents.
\end{enumerate}

\subsubsection*{RLS gain and update via joint Gaussian conditioning}
The unknown row $\bm{w}^{(k)}$ is Gaussian and $\bm{z}_i = \bm{M}_{\mathrm{\tau}}\bm{B}\bm{u}_i + \boldsymbol{\varepsilon}_i$ is a linear
function of it plus independent Gaussian noise, so $(\bm{w}^{(k)}, \bm{z}_i)$ are jointly Gaussian and
the posterior for row $k$ is available in closed form:
\begin{align*}
    \bm{w}^{(k)} \mid \bm{z}_i \;\sim\; \mathcal{N}\Big(
        \hat{\bm{w}}_{i-1}^{(k)} + \bm{\Gamma}_i^{(k)}\bm{S}_i^{-1}\big(\bm{z}_i - \bm{M}_{\mathrm{\tau}}\hat{\bm{B}}_{i-1}\bm{u}_i\big),
        \;\;
        \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)} - \bm{\Gamma}_i^{(k)}\bm{S}_i^{-1}\bm{\Gamma}_i^{(k)\top}
    \Big),
\end{align*}
with the prior mean $\hat{\bm{w}}_{i-1}^{(k)}$ and covariance $\hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}$, the predictive mean $\mathbb{E}[\bm{z}_i] = \bm{M}_{\mathrm{\tau}}\hat{\bm{B}}_{i-1}\bm{u}_i$ and covariance $\bm{S}_i = \bm{D}_i + \bm{R}_i$ from above, and the cross-covariance
$\bm{\Gamma}_i^{(k)} := \mathrm{Cov}(\bm{w}^{(k)}, \bm{z}_i) \in \mathbb{R}^{N\times K}$.
The posterior mean and covariance are $\hat{\bm{w}}_{i}^{(k)}$ and $\hat{\bm{\Sigma}}_{B_i}^{(k)}$.

\paragraph{Cross-covariance.}

The $l$-th column of $\bm{\Gamma}_i^{(k)}$ is
\begin{align*}
    \mathrm{Cov}
    \left(
        \bm{w}^{(k)},
        [\bm{M}_{\mathrm{\tau}}\bm{B}\bm{u}_i]_l
        +
        \varepsilon_{i,l}
    \right).
\end{align*}

Expanding coordinate $l$ of the propagated perturbation as a sum over latents, and using
$[\bm{B}\bm{u}_i]_j = \bm{w}^{(j)\top}\bm{u}_i$,
\begin{align*}
    [\bm{M}_{\mathrm{\tau}}\bm{B}\bm{u}_i]_l
    &= \sum_{j=1}^{K} [\bm{M}_{\mathrm{\tau}}]_{lj}\,[\bm{B}\bm{u}_i]_j
     = \sum_{j=1}^{K} [\bm{M}_{\mathrm{\tau}}]_{lj}\,\bm{w}^{(j)\top}\bm{u}_i.
\end{align*}
The noise is independent of $\bm{B}$, so $\mathrm{Cov}(\bm{w}^{(k)},\varepsilon_{i,l}) = \bm{0}$, and
the rows are independent a priori, so $\mathrm{Cov}(\bm{w}^{(k)},\bm{w}^{(j)}) = \bm{0}$ for
$j \neq k$. Only the $j=k$ term survives:
\begin{align*}
    \mathrm{Cov}\big(\bm{w}^{(k)}, z_{i,l}\big)
    &= \sum_{j=1}^{K} [\bm{M}_{\mathrm{\tau}}]_{lj}\,
       \mathrm{Cov}\big(\bm{w}^{(k)},\, \bm{w}^{(j)\top}\bm{u}_i\big) \\
    &= [\bm{M}_{\mathrm{\tau}}]_{lk}\,\mathrm{Cov}\big(\bm{w}^{(k)}\big)\,\bm{u}_i
     = [\bm{M}_{\mathrm{\tau}}]_{lk}\,\bm{v}_k,
    \qquad
    \bm{v}_k := \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}\bm{u}_i \in \mathbb{R}^{N}.
\end{align*}
Every column is therefore the same vector $\bm{v}_k$, scaled by the corresponding entry of the
$k$-th \emph{column} of $\bm{M}_{\mathrm{\tau}}$. Stacking $l = 1,\dots,K$ gives
$\bm{\Gamma}_i^{(k)} = \bm{v}_k\,(\bm{M}_{\mathrm{\tau}}\bm{e}_k)^\top$, with dimensions
$(N\times N)(N\times 1)(1\times K)(K\times K) = N\times K$.

Written out, the cross-covariance is
\begin{align*}
    \bm{\Gamma}_i^{(k)}
    &=
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    \bm{u}_i
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \in\mathbb{R}^{N\times K}.
\end{align*}


\paragraph{Innovation, mean update, and gain}

The innovation is the difference between the trial's observed perturbation and the predicted perturbation from the previous trial $i-1$. Define
\begin{align*}
    \boldsymbol{\eta}_i
    &:=
    \hat{\bm{z}}_i
    -
    \bm{M}_{\mathrm{\tau}}
    \hat{\bm{B}}_{i-1}
    \bm{u}_i
    \in\mathbb{R}^K
\end{align*}

Substituting
$\bm{\Gamma}_i^{(k)}
=
\hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
\bm{u}_i
\bm{e}_k^\top
\bm{M}_{\mathrm{\tau}}^\top$,
\begin{align*}
    \hat{\bm{w}}_i^{(k)}
    &=
    \hat{\bm{w}}_{i-1}^{(k)}
    +
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    \bm{u}_i
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \bm{S}_i^{-1}
    \boldsymbol{\eta}_i.
\end{align*}

Define the projected gain
\begin{align*}
    \bm{g}_{i,k}
    &:=
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    \bm{u}_i
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \bm{S}_i^{-1},
\end{align*}

so that
\begin{align*}
    \boxed{
    \hat{\bm{w}}_i^{(k)}
    =
    \hat{\bm{w}}_{i-1}^{(k)}
    +
    \bm{g}_{i,k}\boldsymbol{\eta}_i
    }.
\end{align*}

\paragraph{Covariance update.}
The covariance update is
\begin{align*}
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    &=
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    -
    \bm{\Gamma}_i^{(k)}
    \bm{S}_i^{-1}
    \bm{\Gamma}_i^{(k)\top}
    \\
    &=
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    -
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    \bm{u}_i
    \left(
        \bm{e}_k^\top
        \bm{M}_{\mathrm{\tau}}^\top
        \bm{S}_i^{-1}
        \bm{M}_{\mathrm{\tau}}
        \bm{e}_k
    \right)
    \bm{u}_i^\top
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}.
\end{align*}

Let
\begin{align*}
    \alpha_{i,k}
    &:=
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \bm{S}_i^{-1}
    \bm{M}_{\mathrm{\tau}}
    \bm{e}_k.
\end{align*}
\begin{align*}
    \boxed{
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    =
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    -
    \alpha_{i,k}
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    \bm{u}_i\bm{u}_i^\top
    \hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}
    }.
\end{align*}
Repeating for every $k=1,\dots,K$ and stacking recovers $\hat{\bm{B}}_i$ directly. The row-independence assumption enters at $\bm{\Gamma}_i^{(k)}$ for the mean update and grabbing $(\bm{S}_i^{-1})_{kk}$ in the covariance update.

\paragraph{Implementation convention.}
The equations above are written in the observation space of $\bm{z}_i$, so
$\bm{M}_{\mathrm{\tau}}$ appears explicitly. The implementation instead premultiplies the
observation and its noise by $\bm{M}_{\mathrm{\tau}}^{-1}$, tracking
$\bm{M}_{\mathrm{\tau}}^{-1}\hat{\bm{z}}_i$ with noise covariance
$\bm{M}_{\mathrm{\tau}}^{-1}\bm{R}_i\bm{M}_{\mathrm{\tau}}^{-\top}$, so no
$\bm{M}_{\mathrm{\tau}}$ appears anywhere in the update. The two are identical: writing
$\bm{S}_i'$ for the predictive covariance in the transformed space,
\begin{align*}
    \bm{S}_i'
    =
    \operatorname{diag}(\sigma^2_{1|\bm{u}_i},\dots,\sigma^2_{K|\bm{u}_i})
    + \bm{M}_{\mathrm{\tau}}^{-1}\bm{R}_i\bm{M}_{\mathrm{\tau}}^{-\top}
    =
    \bm{M}_{\mathrm{\tau}}^{-1}\bm{S}_i\bm{M}_{\mathrm{\tau}}^{-\top},
    \qquad
    \bm{S}_i'^{-1}
    =
    \bm{M}_{\mathrm{\tau}}^\top\bm{S}_i^{-1}\bm{M}_{\mathrm{\tau}},
\end{align*}
so $(\bm{S}_i'^{-1})_{kk} = \alpha_{i,k}$ exactly, and with
$\boldsymbol{\eta}_i' = \bm{M}_{\mathrm{\tau}}^{-1}\boldsymbol{\eta}_i$ the mean update
$\hat{\bm{\Sigma}}_{B_{i-1}}^{(k)}\bm{u}_i\bm{e}_k^\top\bm{S}_i'^{-1}\boldsymbol{\eta}_i'$
reduces to $\bm{g}_{i,k}\boldsymbol{\eta}_i$. Note this equivalence relies on $\bm{D}_i$
carrying the $\bm{M}_{\mathrm{\tau}}\cdot\bm{M}_{\mathrm{\tau}}^\top$ sandwich; pairing a flat
$\bm{D}_i$ with an $\boldsymbol{\eta}_i$ that includes $\bm{M}_{\mathrm{\tau}}$ gives a filter
mismatched by one factor of $\bm{M}_{\mathrm{\tau}}$.

\subsubsection*{Log-determinant with hypothetical $\bm{u}_{i+1}$}
For a hypothetical stimulation input $\bm{u}_{i+1}$, define the
one-trial-ahead predictive covariance of the observed perturbation as
\begin{align*}
    \bm{S}_i(\bm{u}_{i+1})
    &:=
    \operatorname{Cov}
    \left(
        \bm{z}_{i+1}
        \mid
        \mathcal{D}_i,\bm{u}_{i+1}
    \right) \\
    &=
    \bm{M}_{\mathrm{\tau}}
    \operatorname{diag}
    \left(
        \bm{u}_{i+1}^\top
        \hat{\bm{\Sigma}}_{B_i}^{(1)}
        \bm{u}_{i+1},
        \dots,
        \bm{u}_{i+1}^\top
        \hat{\bm{\Sigma}}_{B_i}^{(K)}
        \bm{u}_{i+1}
    \right)
    \bm{M}_{\mathrm{\tau}}^\top
    +
    \bm{R}_i.
\end{align*}

Because
\[
    \bm{z}_{i+1}
    =
    \bm{M}_{\mathrm{\tau}}
    \bm{B}\bm{u}_{i+1}
    +
    \boldsymbol{\varepsilon}_{i+1},
\]
the cross-covariance between the $k$th parameter row and the
observation vector is
\begin{align*}
    \bm{\Gamma}_{i+1}^{(k)}
    &:=
    \operatorname{Cov}
    \left(
        \bm{w}^{(k)},\bm{z}_{i+1}
    \right) \\
    &=
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \bm{u}_{i+1}
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \in\mathbb{R}^{N\times K}.
\end{align*}

The Gaussian conditioning covariance update is therefore
\begin{align*}
    \hat{\bm{\Sigma}}_{B_{i+1}}^{(k)}
    &=
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    -
    \bm{\Gamma}_{i+1}^{(k)}
    \bm{S}_i(\bm{u}_{i+1})^{-1}
    \bm{\Gamma}_{i+1}^{(k)\top} \\
    &=
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    -
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \bm{u}_{i+1}
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \bm{S}_i(\bm{u}_{i+1})^{-1}
    \bm{M}_{\mathrm{\tau}}
    \bm{e}_k
    \bm{u}_{i+1}^\top
    \hat{\bm{\Sigma}}_{B_i}^{(k)}.
\end{align*}

Define
\begin{align*}
    \alpha_{i,k}(\bm{u}_{i+1})
    &:=
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \bm{S}_i(\bm{u}_{i+1})^{-1}
    \bm{M}_{\mathrm{\tau}}
    \bm{e}_k, \\
    \bm{v}_k
    &:=
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \bm{u}_{i+1}.
\end{align*}

Then each row's covariance update is a rank-1 downdate,
\begin{align*}
    \hat{\bm{\Sigma}}_{B_{i+1}}^{(k)}
    =
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    -
    \alpha_{i,k}(\bm{u}_{i+1})
    \bm{v}_k\bm{v}_k^\top.
\end{align*}

Thus, by the matrix determinant lemma,
\begin{align*}
    \det
    \left(
        \hat{\bm{\Sigma}}_{B_{i+1}}^{(k)}
    \right)
    &=
    \left[
        1
        -
        \alpha_{i,k}(\bm{u}_{i+1})
        \bm{v}_k^\top
        \left(
            \hat{\bm{\Sigma}}_{B_i}^{(k)}
        \right)^{-1}
        \bm{v}_k
    \right]
    \det
    \left(
        \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \right) \\
    &=
    \left[
        1
        -
        \bm{e}_k^\top
        \bm{M}_{\mathrm{\tau}}^\top
        \bm{S}_i(\bm{u}_{i+1})^{-1}
        \bm{M}_{\mathrm{\tau}}
        \bm{e}_k
        \,
        \bm{u}_{i+1}^\top
        \hat{\bm{\Sigma}}_{B_i}^{(k)}
        \bm{u}_{i+1}
    \right]
    \det
    \left(
        \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \right),
\end{align*}
where
\begin{align*}
    \bm{v}_k^\top
    \left(
        \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \right)^{-1}
    \bm{v}_k
    =
    \bm{u}_{i+1}^\top
    \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \bm{u}_{i+1}.
\end{align*}

\subsubsection*{Information gain}

Define the per-row information gain for a hypothetical stimulus $\bm{u}$ as
\begin{align*}
    \mathcal{I}_{i,k}(\bm{u})
    &:=
    \log\det
    \left(
        \hat{\bm{\Sigma}}_{B_i}^{(k)}
    \right)
    -
    \log\det
    \left(
        \hat{\bm{\Sigma}}_{B_{i+1}}^{(k)}
    \right) \\
    &=
    -\log\left[
        1
        -
        \bm{e}_k^\top
        \bm{M}_{\mathrm{\tau}}^\top
        \bm{S}_i(\bm{u})^{-1}
        \bm{M}_{\mathrm{\tau}}
        \bm{e}_k
        \,
        \bm{u}^\top
        \hat{\bm{\Sigma}}_{B_i}^{(k)}
        \bm{u}
    \right].
\end{align*}

The relevant precision for row $k$ is therefore
\begin{align*}
    \bm{e}_k^\top
    \bm{M}_{\mathrm{\tau}}^\top
    \bm{S}_i(\bm{u})^{-1}
    \bm{M}_{\mathrm{\tau}}
    \bm{e}_k,
\end{align*}
This integrates the dynamics-induced mixing through $\bm{M}_{\tau}$, observation uncertainty through $\bm{R}_i$, and parameter uncertainty through $\bm{D}_i$, and therefore depends on the candidate stimulus itself through $\bm{S}_i(\bm{u})$.

Although the rows of $\bm{B}$ are independent a priori, conditioning on the vector-valued observation $\bm{z}_i$ generally induces posterior correlations between rows, particularly when $\bm{M}_{\tau}$ mixes latent coordinates or $\bm{R}_i$ is non-diagonal. We therefore maintain only the marginal covariance of each row and discard the posterior cross-row covariances. The resulting D-optimal criterion is an approximate, row-factorized design criterion.
\begin{align*}
    \bm{s}_{i+1}^{*}
    &=
    \arg\max_{r\in\{1,\dots,N-7\}}
    \sum_{k=1}^{K}
    -
    \log\left[
        1
        -
        \bm{e}_k^\top
        \bm{M}_{\mathrm{\tau}}^\top
        \bm{S}_i(\bm{u}_r)^{-1}
        \bm{M}_{\mathrm{\tau}}
        \bm{e}_k
        \,
        \bm{u}_r^\top
        \hat{\bm{\Sigma}}_{B_i}^{(k)}
        \bm{u}_r
    \right],
    \\
    \bm{u}_r
    &:=
    \operatorname{softplus}
    \left(
        \bm{V}\bm{s}_r+\bm{b}
    \right),
\end{align*}
where
\begin{align*}
    \bm{s}_r
    &:=
    \sum_{j=0}^{7}\bm{e}_{r+j},
    \qquad
    r\in\{1,\dots,N-7\}.
\end{align*}

\subsection*{Choosing the next stimulation pattern $\bm{s}_{i+1}$}
\phantomsection{\label{sec:stim_selection}}

Both criteria reduce to maximizing a set function of the binary pattern $\bm{s}\in\{0,1\}^N$
subject to a sparsity budget, with the nonlinearity $\bm{u} = \mathrm{softplus}(\bm{V}\bm{s}+\bm{b})$
in between:
\begin{align*}
    \bm{s}_{i+1}^{*} = \arg\max_{\|\bm{s}\|_0 \le 8}\; f(\bm{u}(\bm{s})),
    \qquad
    f_{\mathrm{D}} = \sum_{k}\log\big(1+\rho_{i,k}\big),
    \qquad
    f_{\mathrm{A}} = \mathrm{tr}\big(\bm{S}^{-1}\bm{J}^\top\bm{J}\big).
\end{align*}
Exhaustive search is $\binom{N}{8}$ and out of reach, so:
\begin{itemize}[itemsep=0em, topsep=0.2em]
    \item \textbf{Greedy}: add the single channel with the largest marginal gain, recompute, repeat 8 times. $\log\det$ objectives of this form are submodular in the selected set, so greedy carries the standard $(1-1/e)$ guarantee; the trace objective does not in general.
    \item \textbf{Projected / proximal descent}: relax $\bm{s}$ to $[0,1]^N$ and descend, projecting to the sparse binary set either at the end (projected) or at each step (proximal).
    \item Both need the $\bm{V}$ spread matrix and softplus in the chain rule, since neighbouring channels are not independent knobs.
\end{itemize}
The gradient of $f_{\mathrm{D}}$ with respect to $\bm{u}$ is available in closed form,
$\nabla_{\bm{u}} f_{\mathrm{D}} = \sum_k \frac{2\,\hat{\bm{\Sigma}}_{B_i}^{(k)}\bm{u}}{\bm{u}^\top\hat{\bm{\Sigma}}_{B_i}^{(k)}\bm{u} + r_{i,k}}$,
which makes the relaxed problem cheap.

\subsubsection*{Superseded: earlier precision-form sketch}
\inlinecomment{Kept for reference. This version treated the information increment as $\beta\,\mathbb{E}[\Delta]\mathbb{E}[\Delta]^\top$, which measures the size of the expected perturbation rather than the reduction in posterior uncertainty, and needed a free scaling $\beta$. Superseded by the two criteria above.}
For adaptive selection, before each trial $i+1$ we want $\bm{u}_{i+1}$ maximizing the expected
information gain about $\bm{B}$, i.e.\ minimizing the determinant of the covariance of $\bm{B}$
after trial $i+1$. Writing $\boldsymbol{\Delta}_{i+1} = \bm{B}\bm{u}_{i+1}$ for the induced latent
perturbation and $\beta$ for how the data is scaled relative to the prior, the precision form is
\begin{equation*}
    \boldsymbol{\Sigma}_{B_{i+1|i}}^{-1} = \boldsymbol{\Sigma}_{B_{i|i}}^{-1}
    + \beta \; \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{u}_{i+1}]
               \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{u}_{i+1}]^\top,
\end{equation*}
and the matrix determinant lemma plus $\mathbb{E}[\boldsymbol{\Delta}_{i+1}\mid\bm{u}_{i+1}] = \hat{\bm{B}}_i\bm{u}_{i+1}$
gives the objective
\begin{align*}
    \arg\max_{\bm{u}}\;\beta\,
    \bm{u}_{i+1}^\top \hat{\bm{B}}_{i}^\top
    \boldsymbol{\Sigma}_{i}^{B}\,
    \hat{\bm{B}}_{i}\bm{u}_{i+1}.
\end{align*}
