\section*{Problem Formulation}
\subsection*{There are a few ways we can formulate the question}
\begin{enumerate}[itemsep=0em, topsep=0.2em]
    \item Similar to Draelos 2020 first identify the binary connections using group testing, then use the identified connections to fit a weighted model.
    \item Directly use a linear-Poisson model to identify weights using group testing. (Works on the assumption that connections are linear)
    \item If not linear, use a neural network or some other nonlinear approach.
\end{enumerate}
We will focus on (2) first, with fixed continuous stimulation parameters (e.g., current, frequency...)

\section*{Goal}
To develop a method for adaptive stimulation that iteratively identifies the most informative stimulation sites. The dynamics of the reach is modeled using a linear-Poisson model , using a weighted linear-Poisson model to identify the most informative stimulation sites and improve the accuracy of the model.

\subsection*{Dynamics model}
We model stimulation as a perturbation to a latent dynamical system through the mapping $\bm{B}$. Specifically, we model the system using a Poisson linear dynamical system (PLDS), which combines firing rate estimation with dimensionality reduction. We assume linear, Markovian (AR(1)) state evolution, Poisson emissions, and stationary $\bm{A}$. Eventually, we can have a Kalman-like update to allow for nonstationarity in $\bm{C}$\phantomsection\label{sec:nonstationarity} and a compositional model for $\bm{A}$ \phantomsection\label{sec:compositional_model}.

\subsubsection*{State space equation}
\vspace{-0.8cm}
\begin{align*}
    \bm{x}_0 &\sim \mathcal{N}(\boldsymbol{\mu}_0, \boldsymbol{\Sigma}_0) \\
    \bm{x}_t &= \bm{A}\bm{x}_{t-1} + \bm{B}\,\text{softplus}\!\left(\bm{V}\bm{s}_{t_{stim}} + \bm{b}\right) + \boldsymbol{\epsilon}_t, \quad \boldsymbol{\epsilon}_t \sim \mathcal{N}(\bm{0}, \bm{Q}) \\
    \bm{y}_t &\sim \text{Poisson}\!\left(\text{softplus}\!\left(\bm{C}\bm{x}_t + \bm{d}\right)\right)
\end{align*}

\section*{Overview}
\begin{enumerate}
    \item \textbf{Control trials}: Monkey performs control reaches to the target
    \item We fit the PLDS to the control reaches to obtain $\hat{\bm{A}}_{\mathrm{ctrl}}, \hat{\bm{C}}_{\mathrm{ctrl}}, \hat{\bm{d}}_{\mathrm{ctrl}}, \hat{\bm{Q}}_{\mathrm{ctrl}}$
    \item \textbf{Stim trials}: During each trial, perform online filtering using the fitted parameters $\hat{\bm{\theta}}_{\mathrm{i-1}}$,  to get $\{\bm{x}_{i,1:t}\}$ in real time. This gives us the state at which the stimulation happens.
    \begin{enumerate}
        \item At the first trial, we initialize with $\hat{\bm{\theta}}_{\mathrm{i-1}} = \{\hat{\bm{A}}_{\mathrm{ctrl}}, \hat{\bm{B}}_{0|0}, \hat{\bm{b}}_{0|0}, \hat{\bm{C}}_{\mathrm{ctrl}}, \hat{\bm{d}}_{\mathrm{ctrl}}, \hat{\bm{Q}}_{\mathrm{ctrl}}, \hat{\bm{\mu}}_{0}, \hat{\bm{\Sigma}}_{0}\}$, where $\hat{\bm{B}}_{0|0} = \bm{0}$, $\hat{\bm{\Sigma}}_{B_{0|0}} = \sigma_B^2\,\bm{I}$, and $\hat{\bm{b}}_{0|0} = \bm{0}$.
        \item This means that we have no information about the stim-to-latent mapping or the perturbation threshold and the first stimulation sites are chosen at random.
    \end{enumerate}
    \item \hyperref[sec:model_fitting]{\textbf{Model fitting}}: Between each trial, we fit the spike counts $\{\bm{y}_{i,1:T}\}$ to the model using the Extended Kalman Filter (EKF)-style integral and stochastic gradient descent (SGD). This gives us updated parameters $\hat{\bm{\theta}}_{\mathrm{i}} \leftarrow \hat{\bm{\theta}}_{\mathrm{i-1}}$
    \begin{enumerate}
        \item The previous trial would "warm-start" the convergence of the next trial, so we only need to run a few iterations of EKF and SGD between each trial to get a converging estimate of the parameters.
    \end{enumerate}
    \item \hyperref[sec:acquisition]{\textbf{Acquisition:}} Choose the next stimulation site $\bm{s}_{i+1}$ using the covariance estimate of $\hat{\bm{B}}_{i|i}$ ($\hat{\bm{\Sigma}}_{B_{i|i}}$), which is the uncertainty in the stim-to-latent mapping after trial $i$.
    \item Perform the next stimulation and repeat 3-5.
\end{enumerate}

\subsection*{Online forward filtering during a trial}
\phantomsection{\label{sec:online_filtering}}
During a trial, we estimate the latent state $\bm{x}_{i,t}$ in real time using the EKF forward pass with the current parameter estimates $\hat{\boldsymbol{\theta}}_{i-1}$. This gives us an estimate of the latent state in real time.
\subsection*{EKF forward pass} %  (Ignore Ds for now)
The marginal likelihood of the latent trajectory is:
\begin{align*}
    \log p(\bm{y}_{1:T}|\boldsymbol{\theta})
        &= \log \int p(\bm{y}_{1:T} \mid \bm{x}_{1:T};\boldsymbol{\theta})\,
                       p(\bm{x}_{1:T};\boldsymbol{\theta})\, d\bm{x}_{1:T}\\
\bm{y}_t &\sim \text{Poisson}\!\left(\text{softplus}\!\left(\bm{C}\bm{x}_t + \bm{d}\right)\right)\\
\boldsymbol{\theta} &= \{\bm{A, B, C, d, \mu_0, \Sigma_0}\}
\end{align*}

Because the Poisson emission $p(\bm{y}_{1:t} \mid \bm{x}_{1:t};\boldsymbol{\theta})$
is non-Gaussian, the integral is intractable. At time $t-1$, we need to approximate it by linearizing the emission around $\hat{\bm{x}}_{t|t-1}$, the current filtered mean (latent state), yielding a Gaussian approximation to the predictive distribution
\begin{align*}
    p(\bm{y}_t \mid \bm{y}_{1:t-1};\boldsymbol{\theta})
        \approx \mathcal{N}(\boldsymbol{\lambda}_{t|t-1},\, \bm{S}_t), \qquad
    \boldsymbol{\lambda}_{t|t-1} = \text{softplus}\!\left(\bm{C}\hat{\bm{x}}_{t|t-1} + \bm{d}\right)
\end{align*}
where $\bm{S}_t$ is the estimated covariance.
\begin{align*}
    \mathcal{L}(\boldsymbol{\theta})
        \approx \sum_{t=1}^{T} \log \mathcal{N}(\bm{y}_t|\,
                \boldsymbol{\lambda}_{t|t-1},\, \bm{S}_t)
\end{align*}

\subsection*{Emission Jacobian}
In the EKF formulation, we approximate the observation operator $\bm{H}_t$ with the Emission Jacobian described here:
For $\boldsymbol{\lambda}_t = \text{softplus}(\bm{C}\bm{\hat{x}}_t + \bm{d})$, we have the Jacobian (log-rate with respect to the latent state) is:
\begin{align*}
    \bm{H}_t &= \frac{\partial \boldsymbol{\lambda}_t}{\partial \bm{x}_t}\Bigg|_{\bm{\hat{x}}_t}
                 = \text{diag}(\sigma(\bm{C}\bm{\hat{x}}_t + \bm{d}))\,\bm{C}
                 \in \mathbb{R}^{N \times K}
\end{align*}
where $\sigma(\cdot)$ is the sigmoid function (derivative of softplus). We calculate the Emission Jacobian $\bm{H}_t$ to linearize the non-Gaussian Poisson emission and plug into the Kalman equations. This propagates the Gaussian estimations of latent states $\bm{x}_t$ one time step forward each, which we can plug in the Kalman equations.
\subsection*{EKF forward pass prediction update}
\noindent
\begin{minipage}[t]{0.47\textwidth}
\begin{align*}
    \bm{\hat{x}}_{t|t-1} &= \bm{A}\bm{\hat{x}}_{t-1|t-1} 
    + \bm{B}\,\text{softplus}\!\left(\bm{V}\bm{s}_t + \bm{b}\right) \\
    \bm{P}_{t|t-1}       &= \bm{A}\bm{P}_{t-1|t-1}\bm{A}^\top + \bm{Q} \\
    \bm{S}_t             &= \bm{H}_t \bm{P}_{t|t-1} \bm{H}_t^\top
                                + \text{diag}(\boldsymbol{\lambda}_{t|t-1}) \\
    \bm{K}_t             &= \bm{P}_{t|t-1}\bm{H}_t^\top \bm{S}_t^{-1} \\
    \bm{\hat{x}}_{t|t}   &= \bm{\hat{x}}_{t|t-1}
                                + \bm{K}_t(\bm{y}_t - \boldsymbol{\lambda}_{t|t-1}) \\
    \bm{P}_{t|t}         &= (\bm{I} - \bm{K}_t \bm{H}_t)\bm{P}_{t|t-1}
\end{align*}
\end{minipage}
\hfill\vline\hfill
\begin{minipage}[t]{0.47\textwidth}
\begin{itemize}[leftmargin=*, itemsep=0em, topsep=0.2em]
    \item $\bm{\hat{x}}_{t|t-1} \in \mathbb{R}^{K}$: predicted latent state mean
    \item $\bm{\hat{x}}_{t|t} \in \mathbb{R}^{K}$: filtered latent state mean
    \item $\bm{P}_{t|t-1} \in \mathbb{R}^{K \times K}$: predicted latent state covariance
    \item $\bm{P}_{t|t} \in \mathbb{R}^{K \times K}$: filtered latent state covariance
    \item $\bm{S}_t \in \mathbb{R}^{N \times N}$: innovation covariance
    \item $\bm{K}_t \in \mathbb{R}^{K \times N}$: Kalman gain
    \item $(\bm{y}_t - \boldsymbol{\lambda}_{t|t-1})$: innovation (residual)
    \item $\bm{Q} \in \mathbb{R}^{K \times K}$: latent process noise covariance
    \item $\text{diag}(\boldsymbol{\lambda}_{t|t-1})$: Poisson observation noise
    \item $\bm{H}_t \in \mathbb{R}^{N \times K}$: locally linear approximation of emission matrix
\end{itemize}
\end{minipage}

\subsection*{Offline parameter update (Outline)}
At the end of each trial $i$, the following steps are performed in order:
\begin{enumerate}
    \item ``E step" \textbf{EKF forward pass + RTS smoother}:
    Run the EKF forward over $\bm{y}_{i,1:T}$ with $\hat{\boldsymbol{\theta}}_{i-1}$ fixed, then run the RTS smoother backward to get filtered means and covariances $\{\bm{\hat{x}}_{i,1:T},\, \bm{P}_{i,1:T}\}$. $\{\bm{\hat{x}}_{i,t|t},\, \bm{P}_{i,t|t}\}_{t=1}^{T}$ is the lag-one cross-covariance needed for the M-steps below.
    \item ``M step" \textbf{Factorized estimation of parameters}: \phantomsection\label{sec:factorized_estimation} We factorize the estimation of parameters to get point estimates of parameters with no closed form ($\hat{\bm{C}}_i$, $\hat{\bm{d}}_i$, $\hat{\bm{b}}_i$), but still getting covaraince estimates for those that we can using EM. This also keeps the \href{https://probml.github.io/dynamax/notebooks/hmm/casino_hmm_learning.html#compare-the-learning-algorithms}{faster convergence rate of the EM algorithm}.
    \begin{enumerate}
        \item \hyperref[sec:closed_form_update]{\textbf{Closed-form update $\hat{\bm{A}}_i$, $\hat{\bm{Q}}_i$}}
        \item \hyperref[sec:sgd_parameter_update]{\textbf{SGD update $\hat{\bm{C}}_i$, $\hat{\bm{d}}_i$, $\hat{\bm{b}}_i$}}
            Take one gradient step on the single-trial EKF marginal log-likelihood
            $\mathcal{L}_i = \sum_{t=1}^{T} \log\mathcal{N}(\bm{y}_{i,t} \mid
            \boldsymbol{\lambda}_{i,t|t-1}, \bm{S}_{i,t})$:
            \begin{align*}
                (\hat{\bm{C}}_i,\, \hat{\bm{d}}_i)
                    \leftarrow (\hat{\bm{C}}_{i-1},\, \hat{\bm{d}}_{i-1})
                    - \eta_i\, \nabla_{\bm{C},\bm{d}}\,
                    \bigl[-\mathcal{L}_i(\hat{\boldsymbol{\theta}}_{i-1})\bigr]
            \end{align*}
        \item \hyperref[sec:update_B_and_Sigma_B]{\textbf{Update $\hat{\bm{B}}_i$} and $\boldsymbol{\hat\Sigma}_{B_{i|i}}$:} Using the updated $\hat{\bm{A}}_i$, $\hat{\bm{Q}}_i$, $\hat{\bm{C}}_i$, and $\hat{\bm{d}}_i$ from steps 2-3, we update $\hat{\bm{B}}_i$ and get $\boldsymbol{\hat\Sigma}_{B_{i|i}}$
    \end{enumerate}
    \item \textbf{Repeat 1 and 2 until convergence}:
    \begin{enumerate}
        \item Steps (a)-(c) are handled in the DYNAMAX package using conditional moments gaussian smoother (CMGS), but we lay out the details below.
        \item We may need \href{https://probml.github.io/dynamax/notebooks/hmm/casino_hmm_learning.html#hyperparameter-tuning}{hyperparameter optimization} on the batch size and learning rate.
        \item This process can terminate early if the next trial happens before convergence.
        \item We will need to analyze our \textbf{convergence criterion}\phantomsection\label{sec:convergence_criterion} and if we need an algorithm that is quicker / more accurate. (Low rank prior of loading matrix, instead of a state-space formulation) %we may consider online variational inference or ..
    \end{enumerate}
\end{enumerate}

\subsection*{EKF forward pass + RTS smoother}
The EKF forward pass is identical to the online version. 
(see \hyperref[sec:online_filtering]{\textbf{Online forward filtering}})

\subsubsection*{RTS smoother backward pass}
\begin{align*}
    \bm{G}_t             &= \bm{P}_{t|t}\bm{A}^\top \bm{P}_{t+1|t}^{-1} \\
    \hat{\bm{x}}_{t|T}   &= \hat{\bm{x}}_{t|t}
                                + \bm{G}_t\!\left(\hat{\bm{x}}_{t+1|T} - \hat{\bm{x}}_{t+1|t}\right) \\
    \bm{P}_{t|T}         &= \bm{P}_{t|t}
                                + \bm{G}_t\!\left(\bm{P}_{t+1|T} - \bm{P}_{t+1|t}\right)\bm{G}_t^\top
\end{align*}
initialized at $\hat{\bm{x}}_{T|T}$, $\bm{P}_{T|T}$ and run backward for $t = T-1, \ldots, 1$.
\subsection*{EM closed-form update $\hat{\bm{A}}_i$, $\hat{\bm{Q}}_i$}
\phantomsection\label{sec:closed_form_update}
Given the RTS-smoothed statistics $\{\hat{\bm{x}}_{i,t|T},\, \bm{P}_{i,t|T},\, \bm{P}_{i,t,t-1|T}\}_{t=1}^{T}$
from the E step, $\hat{\bm{A}}_i$ and $\hat{\bm{Q}}_i$ admit closed-form M-step updates
identical to those of a standard linear-Gaussian LDS, since the state transition
\begin{align*}
    \bm{x}_{i,t} = \bm{A}\bm{x}_{i,t-1} + \bm{B}\,\text{softplus}\!\left(\bm{V}\bm{s}_{i,t} + \bm{b}\right) + \boldsymbol{\epsilon}_t
\end{align*}
is linear-Gaussian in $\bm{x}_{i,t}$ given the (currently fixed) stimulation input. Only the
emission $\bm{y}_{i,t} \mid \bm{x}_{i,t}$ is non-Gaussian, so it does not enter this update.

\paragraph{Expected complete-data log-likelihood.}
The terms of the complete-data log-likelihood involving $\bm{A}$, $\bm{Q}$ are
\begin{align*}
    \mathbb{E}\!\left[\log p(\bm{x}_{i,1:T})\right]
        \;\dot=\; -\frac{1}{2}\sum_{t=2}^{T}
            \mathbb{E}\!\left[
                \left(\bm{x}_{i,t} - \bm{A}\bm{x}_{i,t-1}\right)^\top
                \bm{Q}^{-1}
                \left(\bm{x}_{i,t} - \bm{A}\bm{x}_{i,t-1}\right)
            \right]
        - \frac{T-1}{2}\log\det\bm{Q}
\end{align*}
where the expectation is taken under the smoothing posterior $p(\bm{x}_{i,1:T} \mid \bm{y}_{i,1:T}, \hat{\boldsymbol{\theta}}_{i-1})$,
and $\dot=$ denotes equality up to additive constants independent of $\bm{A}, \bm{Q}$.

\paragraph{Sufficient statistics.}
Define the smoothed second-moment sums
\begin{align*}
    \bm{S}_{xx}     &= \sum_{t=2}^{T}\left(\bm{P}_{i,t-1|T} + \hat{\bm{x}}_{i,t-1|T}\hat{\bm{x}}_{i,t-1|T}^\top\right) \\
    \bm{S}_{x'x}    &= \sum_{t=2}^{T}\left(\bm{P}_{i,t,t-1|T} + \hat{\bm{x}}_{i,t|T}\hat{\bm{x}}_{i,t-1|T}^\top\right) \\
    \bm{S}_{x'x'}   &= \sum_{t=2}^{T}\left(\bm{P}_{i,t|T} + \hat{\bm{x}}_{i,t|T}\hat{\bm{x}}_{i,t|T}^\top\right)
\end{align*}
where $\bm{S}_{xx}$ uses the "earlier" timestep $t-1$ on both sides, $\bm{S}_{x'x}$ pairs the
"later" timestep $t$ with the earlier $t-1$, and $\bm{P}_{i,t,t-1|T}$ is the lag-one
smoothed cross-covariance from the RTS backward pass above.

\paragraph{M-step.}
Maximizing the expected complete-data log-likelihood with respect to $\bm{A}$ gives a
weighted least-squares solution; setting the gradient to zero yields
\begin{align*}
    \hat{\bm{A}}_i \leftarrow \bm{S}_{x'x}\,\bm{S}_{xx}^{-1}
\end{align*}
Substituting $\hat{\bm{A}}_i$ back in and maximizing with respect to $\bm{Q}$ gives the
residual covariance
\begin{align*}
    \hat{\bm{Q}}_i \leftarrow \frac{1}{T-1}\left(
        \bm{S}_{x'x'} - \hat{\bm{A}}_i\,\bm{S}_{x'x}^\top
    \right)
\end{align*}
which is symmetrized in practice as $\frac{1}{2}(\hat{\bm{Q}}_i + \hat{\bm{Q}}_i^\top)$ to
guard against numerical asymmetry.

\noindent
\textbf{Stim-trial caveat.} If a stimulation pulse falls inside the window $[2,T]$, the
transition $\bm{x}_{i,t_{\mathrm{stim}}} = \bm{A}\bm{x}_{i,t_{\mathrm{stim}}-1} + \bm{B}\,\bm{u}_{i,t_{\mathrm{stim}}} + \boldsymbol{\epsilon}_t$
is no longer purely autonomous, so including $t = t_{\mathrm{stim}}$ in $\bm{S}_{xx}, \bm{S}_{x'x}, \bm{S}_{x'x'}$
would bias $\hat{\bm{A}}_i$ toward absorbing the stim kick. The online, per-trial version of this
update (see \hyperref[sec:acquisition]{Online update loop}) therefore excludes $t = t_{\mathrm{stim}}$
from all three sums; the general form above is the version used when fitting on stim-free
control data ($\hat{\bm{A}}_{\mathrm{ctrl}}$, $\hat{\bm{Q}}_{\mathrm{ctrl}}$).
\subsection*{SGD parameter update $\hat{\bm{C}}_i$, $\hat{\bm{d}}_i$, $\hat{\bm{b}}_i$}
\phantomsection\label{sec:sgd_parameter_update}
Parameters related to the Poisson observation equation ($\hat{\bm{C}}_i$, $\hat{\bm{d}}_i$) and the stim-to-latent threshold ($\hat{\bm{b}}_i$) are updated by differentiating through the EKF pass:
\begin{align*}
    \boldsymbol{\theta}^{(\tau+1)}
        &= \boldsymbol{\theta}^{(\tau)}
           - \eta_\tau \,\nabla_{\boldsymbol{\theta}}\,
             \bigl[-\mathcal{L}(\boldsymbol{\theta}^{(\tau)})\bigr]
\end{align*}
where $\eta_\tau$ is the effective learning rate defined by the Adam optimizer. More specifically:
\begin{align*}
    \bm{m}_{\tau} &= \beta_1 \bm{m}_{\tau-1}
                         + (1-\beta_1)\,\nabla_{\boldsymbol{\theta}}
                         \bigl[-\mathcal{L}(\boldsymbol{\theta}^{(\tau)})\bigr]
                         \tag{momentum} \\
    \bm{v}_{\tau} &= \beta_2 \bm{v}_{\tau-1}
                         + (1-\beta_2)\,\nabla_{\boldsymbol{\theta}}
                         \bigl[-\mathcal{L}(\boldsymbol{\theta}^{(\tau)})\bigr]^2
                         \tag{RMSprop} \\
    \bm{\hat{m}}_{\tau} &= \frac{\bm{m}_{\tau}}{1-\beta_1^\tau}, \qquad
    \bm{\hat{v}}_{\tau}  = \frac{\bm{v}_{\tau}}{1-\beta_2^\tau}
                         \tag{bias correction} \\
    \boldsymbol{\theta}^{(\tau+1)}
                      &= \boldsymbol{\theta}^{(\tau)}
                         - \frac{\alpha}{\sqrt{\bm{\hat{v}}_{\tau}}+\epsilon}\,
                         \bm{\hat{m}}_{\tau}
                         \tag{parameter update}
\end{align*}
where $\bm{m}_\tau$ and $\bm{v}_\tau$ are the first and second moment
estimates, $\beta_1 = 0.9$, $\beta_2 = 0.999$, $\epsilon = 10^{-8}$ are the
default hyperparameters, and $\alpha$ is the base learning rate. Gradients are computed via automatic differentiation in JAX in the DYNAMAX package.

\subsection*{Getting the posterior $p(\bm{\hat{B}}_{i|i-1} \mid \bm{y}_{1:i-1}, \bm{s}_{i})$}
Stimulation pattern $\bm{s}_i \in \{0,1\}^N$ is fixed for trial $i$ and happens at $t_{\mathrm{stim}}$.
\begin{align*}
    \bm{x}_{i,0} &\sim \mathcal{N}(\boldsymbol{\mu}_0, \boldsymbol{\Sigma}_0) \\
    \bm{x}_{i, t} &= \bm{A}\bm{x}_{i, t-1} + \bm{B}\bm{s}_i + \boldsymbol{\epsilon}_t, \quad \boldsymbol{\epsilon}_t \sim \mathcal{N}(\bm{0}, \bm{Q}) \\
    \bm{y}_{i, t} &\sim \text{Poisson}\!\left(\text{softplus}\!\left(\bm{C}\bm{x}_{i, t} + \bm{d}\right)\right)
\end{align*}
We assume we can factorize across the $\boldsymbol{\theta}$s (Much like coordinate ascent variational inference). We get the estimates of $\boldsymbol{\theta} \setminus \bm{B}$ known from PLDS fit via EKF SGD from trials $1, \ldots, i-1$ and treated as fixed at trial $i$. $\bm{s}_{t}$ is known:
\begin{align*}
    \boldsymbol{\theta} \setminus \bm{B} = \{\bm{A},\,\bm{C},\,\bm{d},\,
    \bm{Q},\,\boldsymbol{\mu}_0,\,\boldsymbol{\Sigma}_0\}
    = \hat{\boldsymbol{\theta}}_{i-1} \setminus \bm{B}
\end{align*}

\begin{align*}
    p(\boldsymbol{\theta} \mid \bm{y}_{1:t})
        &\propto p(\bm{y}_{1:t} \mid \boldsymbol{\theta}) \cdot p(\boldsymbol{\theta}) \\
    p(\bm{B} \mid \bm{y}_{1:t}, \bm{s}_{t})
        &\propto p(\bm{y}_{1:t} \mid \bm{B},\, \boldsymbol{\theta} \setminus \bm{B}) \cdot
                 p(\bm{B}) \cdot p(\bm{A}) \cdot p(\bm{C}) \cdots
                 \quad \text{(by independence of } \boldsymbol{\theta} \text{ components)}
\end{align*}

Absorbing $p(\boldsymbol{\theta} \setminus \bm{B})$ into the constant and expanding over $\bm{x}_{1:t}$:
\begin{equation}
    p(\bm{B} \mid \bm{y}_{1:t}, \bm{s}_{t}) \propto 
        \underbrace{p(\bm{B})}_{\text{Initialize via }\mathcal{N}(\bm{0}, \boldsymbol{\Sigma}_0^B)} 
        \int 
        \underbrace{p(\bm{y}_{1:t} \mid \bm{x}_{1:t}, \boldsymbol{\bm{B}, \theta} \setminus \bm{B})}_{\text{Poisson emission}} \;
        \underbrace{p(\bm{x}_{1:t} \mid \bm{s}_{t}, \boldsymbol{\theta})}_{\text{depends on } \bm{B}, \text{can get from EKF}} 
        \, d\bm{x}_{1:t}
       \label{eq:posterior_B}
\end{equation}
Likelihood after marginalizing out $\bm{x}_{1:t}$:
\begin{align*}
    p(\bm{y}_{1:t} \mid \boldsymbol{\theta}) &= \int 
        \underbrace{p(\bm{y}_{1:t} \mid \bm{x}_{1:t}, \boldsymbol{\theta})}_{\text{Poisson emission, need EKF approx.}} \;
        \underbrace{p(\bm{x}_{1:t} \mid \bm{s}_{t}, \boldsymbol{\theta})}_{\text{Gaussian dynamics prior}} 
        \, d\bm{x}_{1:t}
\end{align*}
\begin{align*}
    \boldsymbol{\theta} &= \{\bm{A},\,\bm{B},\,\bm{C},\,
                             \bm{d},\,\bm{Q},\,\boldsymbol{\mu}_0,\,\boldsymbol{\Sigma}_0\}
\end{align*}

Bayes':
\begin{align*}
    \underbrace{p(\boldsymbol{\theta} \mid \bm{y}_{1:t})}_{\text{posterior}} &= \frac{
        \overbrace{p(\bm{y}_{1:t} \mid \boldsymbol{\theta})}^{\text{likelihood}} \cdot
        \overbrace{p(\boldsymbol{\theta})}^{\text{prior}}
    }{
        \underbrace{p(\bm{y}_{1:t})}_{\text{evidence, constant in } \boldsymbol{\theta}}
    }
\end{align*}

\section*{Acquisition function (Selecting next stimulation pattern)}
\phantomsection\label{sec:acquisition}
We need to select the next stimulation pattern based on the updated posterior distribution of $\bm{B}$. Which means we need $\bm{\hat{B}}_{i+1|i}$ and $\boldsymbol{\hat\Sigma}_{B_{i+1|i}}$. We predict what the one timestep ahead prediction after stim for the next trial will be, to get $\boldsymbol{\hat\Sigma}_{B_{i+1|i}}$. Then we can choose stimulation $\bm{s}_{i+1}$ accordingly.

At time step $t-1$, one timestep before stimulation:
\begin{align*}
    \bm{x}_{i+1, \text{t}} &= \bm{A}\bm{x}_{i+1, t-1} + \bm{B}\bm{s}_{i+1} + \boldsymbol{\epsilon}_t\\
    \bm{B}\bm{s}_{i+1} &= \bm{x}_{i+1, \text{t}} - \bm{A}\bm{x}_{i+1, t-1} - \boldsymbol{\epsilon}_t
\end{align*}

Substitute $\bm{\hat{B}}_{i|i}$ as our estimator of $\bm{B}$, and substituting EKF smoothed means, poststim means $\bm{\hat{x}}_{i, t|T}, \bm{\hat{x}}_{i,t-1|T}$, in place of true latent states, and accurate description of estimations:
\begin{align*}
    \bm{\hat{B}}_{i+1|i}\bm{s}_{i+1} &= \bm{\hat{x}}_{i+1,t|T} - \bm{\hat{A}}\bm{\hat{x}}_{i+1,t-1|T} - \boldsymbol{\epsilon}_t
\end{align*}
where:
\begin{itemize}
    \item $\boldsymbol{\epsilon}_t \sim \mathcal{N}(\bm{0}, \bm{Q})$
    \item $ \bm{x}_{t-1} \approx \bm{x}_{t-1|T} \sim \mathcal{N}(\bm{\hat{x}}_{i, t-1|T}, \bm{P}_{i, t-1|T})$
    \item $ \bm{x}_{t} \approx \bm{x}_{t|T} \sim \mathcal{N}(\bm{\hat{x}}_{i, t|T}, \bm{P}_{i, t|T})$
    \item For now we ignore Cov($\bm{\hat{A}}$)
\end{itemize}

Now we want the covariance:
\begin{align*}
    \boldsymbol{\Sigma}_{B_{i+1|i}} = \operatorname{Cov}(\bm{\hat{B}}_{i+1|i}) &= \operatorname{Cov}(\bm{\hat{x}}_{i,t|T}) + \operatorname{Cov}(\bm{\hat{A}}\bm{\hat{x}}_{i,t-1|T}) + \operatorname{Cov}(\boldsymbol{\epsilon}_t)\\
    \boldsymbol{\Sigma}_{B_{i|i}} &\approx \operatorname{Cov}(\bm{\hat{A}}\bm{\hat{x}}_{i,t-1|T} + \bm{\hat{B}}_{i|i}\bm{s}_t + \boldsymbol{\epsilon}_t) + \bm{\hat{A}}\bm{P}_{i,t-1|T}\bm{\hat{A}}^\top + \bm{Q}\\
    \boldsymbol{\Sigma}_{B_{i|i}} &\approx \operatorname{Cov}(\bm{\hat{B}}_{i|i}\bm{s}_t) + 2\bm{\hat{A}}\bm{P}_{i,t-1|T}\bm{\hat{A}}^\top + 2\bm{Q}
\end{align*}

Let $\bm{\hat{B}}_{i|i}\bm{s}_t$ be $\bm{Bs}$ for now.
\begin{align*}
    \operatorname{Cov}(\bm{Bs}) &= \mathbb{E}\!\left[\bm{Bss}^T\bm{B}^T\right] - \mathbb{E}[\bm{Bs}]\,\mathbb{E}[\bm{Bs}]^T
\end{align*}

To capture the dependencies across channels (not IID), use the posterior of $\bm{B}$ from \eqref{eq:posterior_B}. We know from EKF $\mathbb{E}[\bm{B}] = \bm{\hat{B}}_{i|i}, \operatorname{Cov}(\bm{\hat{B}}_{i|i}) = \boldsymbol{\hat\Sigma}_{B_{i|i}}$:
\begin{align*}
    \operatorname{Cov}(\bm{Bs}) 
        &= \mathbb{E}\!\left[\bm{Bss}^T\bm{B}^T\right] - \mathbb{E}[\bm{Bs}]\,\mathbb{E}[\bm{Bs}]^T \\
    \operatorname{Cov}(\bm{Bs}) 
        &= \mathbb{E}\!\left[\bm{B}\right]\bm{ss}^T\mathbb{E}\!\left[\bm{B}^T\right] 
         + \operatorname{Cov}(\bm{B}) \cdot \bm{ss}^T
         - \mathbb{E}[\bm{B}]\bm{ss}^T\mathbb{E}[\bm{B}]^T \\
    \operatorname{Cov}(\bm{\hat{B}}_{i|i}\,\bm{s}_t)
        &= \boldsymbol{\hat\Sigma}_{B_{i|i}}\, \bm{s}_t\bm{s}_t^\top
\end{align*}

Substituting back:
\begin{align*}
    \boldsymbol{\hat\Sigma}_{B_{i|i}}
        &\approx \boldsymbol{\hat\Sigma}_{B_{i|i}}\, \bm{s}_t\bm{s}_t^\top
         + 2\bm{\hat{A}}\bm{P}_{i,t-1|T}\bm{\hat{A}}^\top + 2\bm{Q}
\end{align*}

\subsubsection*{Derivation}
For adaptive selection of stimulation patterns, before each trial $i+1$, we want to select 
$\bm{s}_{i+1}$ that maximizes the expected information gain about $\bm{B}$. 
Equivalently, this would be selecting $\bm{s}_{i+1}$ that minimizes the determinant 
of the covariance of $\bm{B}$ after trial $i+1$ ($\boldsymbol{\hat\Sigma}_{B_{i+1|i+1}}$).
\begin{align*}
    \text{Let } & \boldsymbol{\Sigma}_{B_{i|i}}^{-1}
    && \text{be the precision matrix of $\bm{B}$ after trial $i$}\\
    \text{Let } & \boldsymbol{\Delta}_{i+1} = \bm{B}\bm{s}_{i+1}
    && \text{be the latent perturbation induced by stimulation pattern $\bm{s}_{i+1}$}\\
    \text{Let } & \beta
    && \text{be how we scale our data relative to our prior (could be $1/n$)}
\end{align*}
In terms of precision, we can formulate the Gaussian case as:
\begin{equation*}
    \boldsymbol{\Sigma}_{B_{i+1|i}}^{-1} = \boldsymbol{\Sigma}_{B_{i|i}}^{-1}
    + \beta \; \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}] 
               \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]^\top
\end{equation*}
Taking the determinant and applying the matrix determinant lemma:
\begin{align*}
    \det\boldsymbol{\Sigma}_{B_{i+1|i}}^{-1} 
    &= \det\!\Big(\boldsymbol{\Sigma}_{B_{i|i}}^{-1}
       + \beta\,\mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]
                \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]^\top\Big) \\
    &= \Big(1 + \beta\,\mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]^\top
       \boldsymbol{\Sigma}_{i}^{B}\,
       \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]\Big)
       \det\!\big(\boldsymbol{\Sigma}_{B_{i|i}}^{-1}\big)
\end{align*}
Since we minimize over $\bm{s}_{i+1}$, the objective function is therefore:
\begin{align*}
    \arg\max_{\bm{s}}\;\det\big(\boldsymbol{\Sigma}_{B_{i+1|i}}^{-1}\big)
    &= \arg\max_{\bm{s}}\;\Big(1 + \beta\,\mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]^\top
       \boldsymbol{\Sigma}_{i}^{B}\,
       \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]\Big)
       \det\!\big(\boldsymbol{\Sigma}_{B_{i|i}}^{-1}\big) \\
    &= \arg\max_{\bm{s}}\;\beta\,\mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]^\top
       \boldsymbol{\Sigma}_{i}^{B}\,
       \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}]
\end{align*}
We can further expand out the expected perturbation in latent space:
\begin{align*}
    \mathbb{E}[\boldsymbol{\Delta}_{i+1} \mid \bm{s}_{i+1}] 
    = \bm{\hat{B}}_{i}\bm{s}_{i+1}
\end{align*}
where $\bm{\hat{B}}_{i}$ is estimated from trials $1,\ldots,i$, giving the 
final objective:
\begin{align*}
    \arg\max_{\bm{s}}\;\beta\,
    \bm{s}_{i+1}^\top \bm{\hat{B}}_{i}^\top
    \boldsymbol{\Sigma}_{i}^{B}\,
    \bm{\hat{B}}_{i}\bm{s}_{i+1}
\end{align*}

\subsection*{Online update loop (trial $i \geq 1$)}

At the end of each trial $i$, the following steps are performed in order:

\begin{enumerate}

\item \textbf{EKF forward pass + RTS smoother.}
Run the EKF forward over $\bm{y}_{i,1:T}$ with $\hat{\boldsymbol{\theta}}_{i-1}$ fixed,
then run the RTS smoother backward:
\begin{align*}
    \text{Filtered:} \quad  &\hat{\bm{x}}_{i,t|t} = \hat{\bm{x}}_{i,t|t-1}
        + \bm{K}_{i,t}(\bm{y}_{i,t} - \boldsymbol{\lambda}_{i,t|t-1}), \quad
        \bm{P}_{i,t|t} = (\bm{I} - \bm{K}_{i,t}\bm{H}_{i,t})\bm{P}_{i,t|t-1} \\
    \text{Smoothed:} \quad  &\hat{\bm{x}}_{i,t|T} = \hat{\bm{x}}_{i,t|t}
        + \bm{G}_{i,t}(\hat{\bm{x}}_{i,t+1|T} - \hat{\bm{x}}_{i,t+1|t}), \quad
        \bm{P}_{i,t|T} = \bm{P}_{i,t|t}
        + \bm{G}_{i,t}(\bm{P}_{i,t+1|T} - \bm{P}_{i,t+1|t})\bm{G}_{i,t}^\top
\end{align*}
where $\bm{G}_{i,t} = \bm{P}_{i,t|t}\bm{A}^\top\bm{P}_{i,t+1|t}^{-1}$
is the smoother gain and $\bm{P}_{i,t,t-1|T} = \bm{P}_{i,t|t}\bm{G}_{i,t-1}^\top$
is the lag-one cross-covariance needed for the M-steps below.

\item \textbf{Closed-form update $\bm{A}$, $\bm{Q}$ (online EM M-step)}
Exclude $t = t_{\mathrm{stim}}$ to avoid absorbing the stim kick into $\bm{A}$:
\begin{align*}
    \hat{\bm{A}}_i
        &\leftarrow \left(\sum_{\substack{t=2 \\ t \neq t_{\mathrm{stim}}}}^{T}
            \bm{P}_{i,t,t-1|T} + \hat{\bm{x}}_{i,t|T}\hat{\bm{x}}_{i,t-1|T}^\top
            \right)
            \left(\sum_{\substack{t=2 \\ t \neq t_{\mathrm{stim}}}}^{T}
            \bm{P}_{i,t-1|T} + \hat{\bm{x}}_{i,t-1|T}\hat{\bm{x}}_{i,t-1|T}^\top
            \right)^{-1} \\[4pt]
    \hat{\bm{Q}}_i
        &\leftarrow \frac{1}{T-2}\sum_{\substack{t=2 \\ t \neq t_{\mathrm{stim}}}}^{T}
            \left[\bm{P}_{i,t|T}
            + \hat{\bm{x}}_{i,t|T}\hat{\bm{x}}_{i,t|T}^\top
            - \hat{\bm{A}}_i\!\left(
            \bm{P}_{i,t,t-1|T}^\top
            + \hat{\bm{x}}_{i,t-1|T}\hat{\bm{x}}_{i,t|T}^\top
            \right)\right]
\end{align*}

\item \textbf{SGD update $\bm{C}$, $\bm{d}$.}
Take one gradient step on the single-trial EKF marginal log-likelihood
$\mathcal{L}_i = \sum_{t=1}^{T} \log\mathcal{N}(\bm{y}_{i,t} \mid
\boldsymbol{\lambda}_{i,t|t-1}, \bm{S}_{i,t})$:
\begin{align*}
    (\hat{\bm{C}}_i,\, \hat{\bm{d}}_i)
        \leftarrow (\hat{\bm{C}}_{i-1},\, \hat{\bm{d}}_{i-1})
        - \eta_i\, \nabla_{\bm{C},\bm{d}}\,
          \bigl[-\mathcal{L}_i(\hat{\boldsymbol{\theta}}_{i-1})\bigr]
\end{align*}
where $\eta_i \propto 1/i$ is a decaying learning rate.

\item \textbf{Update $\bm{B}$ (trial-level Kalman two-step).}
Using the freshly updated $\hat{\bm{A}}_i$, $\hat{\bm{Q}}_i$, $\hat{\bm{C}}_i$,
$\hat{\bm{d}}_i$ from steps 2--3.

\textit{Predict} — propagate $\bm{B}$ uncertainty from trial $i-1$ to $i$, using
the RTS-smoothed covariance $\bm{P}_{i-1,\,t_{\mathrm{stim}}-1|T}$ from the
previous trial (tighter than the filtered covariance since it uses all $T$ observations):
\begin{align*}
    \boldsymbol{\hat\Sigma}_{B_{i|i-1}}
        \leftarrow \boldsymbol{\hat\Sigma}_{B_{i-1|i-1}}\,\bm{s}_i\bm{s}_i^\top
           + 2\,\hat{\bm{A}}_i\,\bm{P}_{i-1,\,t_{\mathrm{stim}}-1|T}\,\hat{\bm{A}}_i^\top
           + 2\,\hat{\bm{Q}}_i
\end{align*}

\textit{Update} — condition on $\bm{y}_{i,\,t_{\mathrm{stim}}}$ via EKF linearization
using $\hat{\bm{C}}_i$, $\hat{\bm{d}}_i$:
\begin{align*}
    \bm{K}_i^B
        &= \boldsymbol{\hat\Sigma}_{B_{i|i-1}}\,\bm{s}_i\,
           \bm{H}_{i,t_{\mathrm{stim}}}^\top\!\left(
           \bm{H}_{i,t_{\mathrm{stim}}}\,\boldsymbol{\hat\Sigma}_{B_{i|i-1}}\,
           \bm{s}_i\bm{s}_i^\top\bm{H}_{i,t_{\mathrm{stim}}}^\top
           + \mathrm{diag}(\boldsymbol{\lambda}_{i,t_{\mathrm{stim}}})^{-1}
           \right)^{-1} \\[4pt]
    \hat{\bm{B}}_{i|i}
        &\leftarrow \hat{\bm{B}}_{i|i-1}
           + \bm{K}_i^B\!\left(\bm{y}_{i,\,t_{\mathrm{stim}}}
           - \boldsymbol{\lambda}_{i,t_{\mathrm{stim}}}\right)\bm{s}_i^\top \\[4pt]
    \boldsymbol{\hat\Sigma}_{B_{i|i}}
        &\leftarrow \left(\bm{I} - \bm{K}_i^B\,
           \bm{H}_{i,t_{\mathrm{stim}}}\,\bm{s}_i\bm{s}_i^\top\right)
           \boldsymbol{\hat\Sigma}_{B_{i|i-1}}
\end{align*}

\item \textbf{Select next stimulation pattern.}
\begin{align*}
    \bm{s}_{i+1}^*
        = \arg\max_{\bm{s}\,\in\,\{0,1\}^N,\;\|\bm{s}\|_0\,=\,k}
          \;\bm{s}^\top \hat{\bm{B}}_{i|i}^\top\,
          \boldsymbol{\hat\Sigma}_{B_{i|i}}\,
          \hat{\bm{B}}_{i|i}\,\bm{s}
\end{align*}


Steps 2 and 3 are decoupled: $\bm{B}$ is updated via the Kalman mechanism and 
$\boldsymbol{\theta} \setminus \bm{B}$ via SGD, both conditioned on the same EKF pass 
from step 1. This introduces an approximation: the EKF in step 1 is run under 
$\bm{\hat{B}}_{i-1|i-1}$, so the filtered states used in step 2 reflect the previous 
trial's $\bm{B}$ estimate rather than the current one.

\end{enumerate}

\subsection*{Posterior for $\bm{B}$ and predict covariance derivation}

We treat $\bm{B}$ as a random variable and factorize across $\boldsymbol{\theta}$
components (analogous to CAVI), treating $\hat{\boldsymbol{\theta}}_{i-1} \setminus
\bm{B}$ as fixed. The posterior after $i-1$ trials is approximated as Gaussian:
\begin{align*}
    p(\bm{B} \mid \bm{y}_{1:i-1}, \bm{s}_{1:i-1})
        \approx \mathcal{N}(\hat{\bm{B}}_{i-1|i-1},\,\boldsymbol{\hat\Sigma}_{B_{i-1|i-1}})
\end{align*}
The Poisson emission is non-Gaussian so each update is approximated via EKF linearization
at $t_{\mathrm{stim}}$, maintaining the Gaussian form across trials.

\paragraph{Derivation of the predict covariance.}
Rearranging the dynamics at $t_{\mathrm{stim}}$ to isolate the $\bm{B}$ contribution:
\begin{align*}
    \hat{\bm{B}}_{i|i-1}\,\bm{s}_i
        &= \hat{\bm{x}}_{i,\,t_{\mathrm{stim}}|t_{\mathrm{stim}}-1}
           - \hat{\bm{A}}_i\,\hat{\bm{x}}_{i,\,t_{\mathrm{stim}}-1|t_{\mathrm{stim}}-1}
           - \boldsymbol{\epsilon}_{t_{\mathrm{stim}}}
\end{align*}
Taking the covariance, using
$\mathrm{Cov}(\hat{\bm{B}}_{i-1|i-1}\bm{s}_i) =
\boldsymbol{\hat\Sigma}_{B_{i-1|i-1}}\bm{s}_i\bm{s}_i^\top$ (derived below)
and ignoring $\mathrm{Cov}(\hat{\bm{A}})$:
\begin{align*}
    \boldsymbol{\hat\Sigma}_{B_{i|i-1}}
        \approx \boldsymbol{\hat\Sigma}_{B_{i-1|i-1}}\,\bm{s}_i\bm{s}_i^\top
        + 2\,\hat{\bm{A}}_i\,\bm{P}_{i-1,\,t_{\mathrm{stim}}-1|T}\,\hat{\bm{A}}_i^\top
        + 2\,\hat{\bm{Q}}_i
\end{align*}
